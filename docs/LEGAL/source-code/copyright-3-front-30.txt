  34  # A-P2-15: 请求体大小限制中间件
  35  class RequestBodySizeLimitMiddleware:
  36      """
  37      请求体大小限制中间件
  38  
  39      防止过大的请求体导致 OOM 或 DoS 攻击。
  40      默认限制 10MB，可通过环境变量 REQUEST_BODY_MAX_BYTES 配置。
  41      对于文件上传接口（如 /api/stylist/v2/tryon），限制放宽到 50MB。
  42      """
  43  
  44      # 默认请求体大小限制（10MB）
  45      DEFAULT_MAX_BODY_BYTES = 10 * 1024 * 1024
  46  
  47      # 文件上传相关路径的放宽限制（50MB）
  48      UPLOAD_MAX_BODY_BYTES = 50 * 1024 * 1024
  49  
  50      # 文件上传路径前缀
  51      UPLOAD_PATH_PREFIXES = (
  52          "/api/stylist/v2/tryon",
  53          "/api/virtual-tryon",
  54          "/api/body-analysis",
  55          "/api/style-analysis",
  56      )
  57  
  58      def __init__(self, app):
  59          self.app = app
  60          import os
  61          self._max_body_bytes = int(
  62              os.getenv("REQUEST_BODY_MAX_BYTES", self.DEFAULT_MAX_BODY_BYTES)
  63          )
  64          self._upload_max_body_bytes = int(
  65              os.getenv("UPLOAD_BODY_MAX_BYTES", self.UPLOAD_MAX_BODY_BYTES)
  66          )
  67  
  68      async def __call__(self, scope, receive, send):
  69          if scope["type"] != "http":
  70              await self.app(scope, receive, send)
  71              return
  72  
  73          # 确定当前路径的大小限制
  74          path = scope.get("path", "")
  75          max_bytes = self._upload_max_body_bytes
  76          for prefix in self.UPLOAD_PATH_PREFIXES:
  77              if path.startswith(prefix):
  78                  break
  79          else:
  80              max_bytes = self._max_body_bytes
  81  
  82          body_size = 0
  83          received_body_parts = []
  84  
  85          async def limited_receive():
  86              nonlocal body_size
  87              message = await receive()
  88  
  89              if message["type"] == "http.request":
  90                  body = message.get("body", b"")
  91                  body_size += len(body)
  92                  received_body_parts.append(body)
  93  
  94                  if body_size > max_bytes:
  95                      # 请求体过大，返回 413
  96                      raise RequestBodyTooLargeError(
  97                          max_bytes=max_bytes,
  98                          actual_bytes=body_size,
  99                      )
 100  
 101              return message
 102  
 103          try:
 104              await self.app(scope, limited_receive, send)
 105          except RequestBodyTooLargeError as e:
 106              # 构造 413 响应
 107              response = JSONResponse(
 108                  status_code=413,
 109                  content={
 110                      "error": "Request Entity Too Large",
 111                      "detail": f"Request body size ({e.actual_bytes} bytes) exceeds limit ({e.max_bytes} bytes)",
 112                      "max_size_bytes": e.max_bytes,
 113                  },
 114              )
 115              await response(scope, receive, send)

  20  # D65 Illuminant Reference Values
  21  # ============================================================

  23  _D65_XN = 0.95047
  24  _D65_YN = 1.00000
  25  _D65_ZN = 1.08883

  28  # ============================================================

  32  def rgb_to_lab(r: int, g: int, b: int) -> Tuple[float, float, float]:
  33      """Convert RGB (0-255) to CIELAB using D65 illuminant.
  34  
  35      Args:
  36          r: Red channel (0-255)
  37          g: Green channel (0-255)
  38          b: Blue channel (0-255)
  39  
  40      Returns:
  41          Tuple of (L*, a*, b*) CIELAB values
  42      """
  43      rn = r / 255.0
  44      gn = g / 255.0
  45      bn = b / 255.0
  46  
  47      def linearize(c: float) -> float:
  48          return ((c + 0.055) / 1.055) ** 2.4 if c > 0.04045 else c / 12.92
  49  
  50      rl = linearize(rn)
  51      gl = linearize(gn)
  52      bl = linearize(bn)
  53  
  54      x = (rl * 0.4124564 + gl * 0.3575761 + bl * 0.1804375) / _D65_XN
  55      y = (rl * 0.2126729 + gl * 0.7151522 + bl * 0.0721750) / _D65_YN
  56      z = (rl * 0.0193339 + gl * 0.1191920 + bl * 0.9503041) / _D65_ZN
  57  
  58      delta = 6.0 / 29.0
  59  
  60      def f(t: float) -> float:
  61          if t > delta ** 3:
  62              return t ** (1.0 / 3.0)
  63          return t / (3.0 * delta * delta) + 4.0 / 29.0
  64  
  65      fx, fy, fz = f(x), f(y), f(z)
  66      l_star = 116.0 * fy - 16.0
  67      a_star = 500.0 * (fx - fy)
  68      b_star = 200.0 * (fy - fz)
  69  
  70      return (l_star, a_star, b_star)
  71

  73  def lab_to_rgb(l: float, a: float, b: float) -> Tuple[int, int, int]:
  74      """Convert CIELAB to RGB (0-255).
  75  
  76      Args:
  77          l: L* value
  78          a: a* value
  79          b: b* value
  80  
  81      Returns:
  82          Tuple of (R, G, B) values (0-255)
  83      """
  84      delta = 6.0 / 29.0
  85      fy = (l + 16.0) / 116.0
  86      fx = a / 500.0 + fy
  87      fz = fy - b / 200.0
  88  
  89      def inv_f(t: float) -> float:
  90          if t > delta:
  91              return t ** 3
  92          return 3.0 * delta * delta * (t - 4.0 / 29.0)
  93  
  94      xn = inv_f(fx) * _D65_XN
  95      yn = inv_f(fy) * _D65_YN
  96      zn = inv_f(fz) * _D65_ZN
  97  
  98      rl = 3.2404542 * xn - 1.5371385 * yn - 0.4985314 * zn
  99      gl = -0.9692660 * xn + 1.8760108 * yn + 0.0415560 * zn
 100      bl = 0.0556434 * xn - 0.2040259 * yn + 1.0572252 * zn
 101  
 102      def gamma(c: float) -> float:
 103          c = max(0.0, c)
 104          return 1.055 * c ** (1.0 / 2.4) - 0.055 if c > 0.0031308 else 12.92 * c
 105  
 106      return (
 107          min(255, max(0, round(gamma(rl) * 255))),
 108          min(255, max(0, round(gamma(gl) * 255))),
 109          min(255, max(0, round(gamma(bl) * 255))),
 110      )

 129  # ============================================================
 130  # CIEDE2000 Color Difference
 131  # ============================================================
 132  
 133  def delta_e_ciede2000(
 134      lab1: Tuple[float, float, float],
 135      lab2: Tuple[float, float, float],
 136      kl: float = 1.0,
 137      kc: float = 1.0,
 138      kh: float = 1.0,
 139  ) -> float:
 140      """Compute CIEDE2000 color difference between two CIELAB colors.
 141  
 142      Args:
 143          lab1: First CIELAB color (L*, a*, b*)
 144          lab2: Second CIELAB color (L*, a*, b*)
 145          kl: Lightness weighting factor (default 1.0)
 146          kc: Chroma weighting factor (default 1.0)
 147          kh: Hue weighting factor (default 1.0)
 148  
 149      Returns:
 150          CIEDE2000 color difference value
 151      """
 152      l1, a1, b1 = lab1
 153      l2, a2, b2 = lab2
 154  
 155      c1 = math.sqrt(a1 ** 2 + b1 ** 2)
 156      c2 = math.sqrt(a2 ** 2 + b2 ** 2)
 157      c_avg = (c1 + c2) / 2.0
 158  
 159      c_avg_7 = c_avg ** 7
 160      g = 0.5 * (1.0 - math.sqrt(c_avg_7 / (c_avg_7 + 25.0 ** 7)))
 161  
 162      a1p = a1 * (1.0 + g)
 163      a2p = a2 * (1.0 + g)
 164  
 165      c1p = math.sqrt(a1p ** 2 + b1 ** 2)
 166      c2p = math.sqrt(a2p ** 2 + b2 ** 2)
 167  
 168      h1p = math.degrees(math.atan2(b1, a1p)) % 360.0
 169      h2p = math.degrees(math.atan2(b2, a2p)) % 360.0
 170  
 171      dl = l2 - l1
 172      dcp = c2p - c1p
 173  
 174      if c1p * c2p == 0:
 175          dhp = 0.0
 176      elif abs(h2p - h1p) <= 180.0:
 177          dhp = h2p - h1p
 178      elif h2p - h1p > 180.0:
 179          dhp = h2p - h1p - 360.0
 180      else:
 181          dhp = h2p - h1p + 360.0
 182  
 183      dhp_val = 2.0 * math.sqrt(c1p * c2p) * math.sin(math.radians(dhp / 2.0))
 184  
 185      lp_avg = (l1 + l2) / 2.0
 186      cp_avg = (c1p + c2p) / 2.0
 187  
 188      if c1p * c2p == 0:
 189          hp_avg = h1p + h2p
 190      elif abs(h1p - h2p) <= 180.0:
 191          hp_avg = (h1p + h2p) / 2.0
 192      elif h1p + h2p < 360.0:
 193          hp_avg = (h1p + h2p + 360.0) / 2.0
 194      else:
 195          hp_avg = (h1p + h2p - 360.0) / 2.0
 196  
 197      t = (1.0
 198           - 0.17 * math.cos(math.radians(hp_avg - 30.0))
 199           + 0.24 * math.cos(math.radians(2.0 * hp_avg))
 200           + 0.32 * math.cos(math.radians(3.0 * hp_avg + 6.0))
 201           - 0.20 * math.cos(math.radians(4.0 * hp_avg - 63.0)))
 202  
 203      sl = 1.0 + 0.015 * (lp_avg - 50.0) ** 2 / math.sqrt(20.0 + (lp_avg - 50.0) ** 2)
 204      sc = 1.0 + 0.045 * cp_avg
 205      sh = 1.0 + 0.015 * cp_avg * t
 206  
 207      cp_avg_7 = cp_avg ** 7
 208      rt = (-math.sin(math.radians(2.0 * (30.0 * math.exp(-((hp_avg - 275.0) / 25.0) ** 2))))
 209            * 2.0 * math.sqrt(cp_avg_7 / (cp_avg_7 + 25.0 ** 7)))
 210  
 211      result = math.sqrt(
 212          (dl / (kl * sl)) ** 2
 213          + (dcp / (kc * sc)) ** 2
 214          + (dhp_val / (kh * sh)) ** 2
 215          + rt * (dcp / (kc * sc)) * (dhp_val / (kh * sh))
 216      )
 217  
 218      return result

 307  def compute_ita(l_star: float, b_star: float) -> float:
 308      """Compute Individual Typology Angle (ITA).
 309  
 310      ITA = arctan((L* - 50) / b*) * (180 / pi)
 311  
 312      Args:
 313          l_star: CIELAB L* value
 314          b_star: CIELAB b* value
 315  
 316      Returns:
 317          ITA value in degrees
 318      """
 319      if abs(b_star) < 1e-10:
 320          return 90.0 if l_star > 50 else -90.0
 321      return math.atan((l_star - 50.0) / b_star) * (180.0 / math.pi)

 324  def compute_chroma(a_star: float, b_star: float) -> float:
 325      """Compute chroma C* = sqrt(a*^2 + b*^2).

 327      Args:
 328          a_star: CIELAB a* value
 329          b_star: CIELAB b* value

 332          Chroma value

 337  def is_skin_pixel_cielab(r: int, g: int, b: int) -> bool:
 338      """Skin pixel detection using CIELAB color space.
 339  
 340      Based on research: human skin in CIELAB falls within:
 341      L* in [20, 90], a* in [2, 20], b* in [5, 35]
 342  
 343      Args:
 344          r: Red channel (0-255)
 345          g: Green channel (0-255)
 346          b: Blue channel (0-255)
 347  
 348      Returns:
 349          True if the pixel is likely a skin pixel
 350      """
 351      l, a, b_val = rgb_to_lab(r, g, b)
 352      if l < 15 or l > 95:
 353          return False
 354      if a < -5 or a > 25:
 355          return False
 356      if b_val < 2 or b_val > 40:
 357          return False
 358      return True

 361  # ============================================================
 362  # Vectorized Batch Conversions (numpy)
 363  # ============================================================
 364  
 365  def rgb_to_lab_batch(pixels: np.ndarray) -> np.ndarray:
 366      """Convert an array of RGB pixels to CIELAB using D65 illuminant (vectorized).
 367  
 368      This is the vectorized equivalent of rgb_to_lab() for batch processing.
 369      Uses the same algorithm and constants, ensuring identical results.
 370  
 371      Args:
 372          pixels: numpy array of shape (N, 3) with RGB values in [0, 255]
 373  
 374      Returns:
 375          numpy array of shape (N, 3) with (L*, a*, b*) CIELAB values
 376      """
 377      r_flat = pixels[:, 0].astype(np.float64) / 255.0
 378      g_flat = pixels[:, 1].astype(np.float64) / 255.0
 379      b_flat = pixels[:, 2].astype(np.float64) / 255.0
 380  
 381      # Vectorized sRGB linearization (same thresholds as scalar version)
 382      rl = np.where(r_flat > 0.04045, ((r_flat + 0.055) / 1.055) ** 2.4, r_flat / 12.92)
 383      gl = np.where(g_flat > 0.04045, ((g_flat + 0.055) / 1.055) ** 2.4, g_flat / 12.92)
 384      bl = np.where(b_flat > 0.04045, ((b_flat + 0.055) / 1.055) ** 2.4, b_flat / 12.92)
 385  
 386      # Linear RGB -> XYZ (D65), same matrix as scalar version
 387      x = (rl * 0.4124564 + gl * 0.3575761 + bl * 0.1804375) / _D65_XN
 388      y = (rl * 0.2126729 + gl * 0.7151522 + bl * 0.0721750) / _D65_YN
 389      z = (rl * 0.0193339 + gl * 0.1191920 + bl * 0.9503041) / _D65_ZN
 390  
 391      # XYZ -> CIELAB, same delta and formula as scalar version
 392      delta = 6.0 / 29.0
 393      fx = np.where(x > delta ** 3, x ** (1.0 / 3.0), x / (3.0 * delta * delta) + 4.0 / 29.0)
 394      fy = np.where(y > delta ** 3, y ** (1.0 / 3.0), y / (3.0 * delta * delta) + 4.0 / 29.0)
 395      fz = np.where(z > delta ** 3, z ** (1.0 / 3.0), z / (3.0 * delta * delta) + 4.0 / 29.0)
 396  
 397      lab = np.column_stack([116.0 * fy - 16.0, 500.0 * (fx - fy), 200.0 * (fy - fz)])
 398      return lab
 399  
 400  
 401  def lab_to_rgb_batch(lab_pixels: np.ndarray) -> np.ndarray:
 402      """Convert an array of CIELAB values to RGB (vectorized).
 403  
 404      This is the vectorized equivalent of lab_to_rgb() for batch processing.
 405      Uses the same algorithm and constants, ensuring identical results.
 406  
 407      Args:
 408          lab_pixels: numpy array of shape (N, 3) with (L*, a*, b*) values
 409  
 410      Returns:
 411          numpy array of shape (N, 3) with RGB values in [0, 255] as int32
 412      """
 413      l = lab_pixels[:, 0]
 414      a = lab_pixels[:, 1]
 415      b = lab_pixels[:, 2]
 416  
 417      delta = 6.0 / 29.0
 418      fy = (l + 16.0) / 116.0
 419      fx = a / 500.0 + fy
 420      fz = fy - b / 200.0
 421  
 422      # Inverse f function (same as scalar version)
 423      xn = np.where(fx > delta, fx ** 3, 3.0 * delta * delta * (fx - 4.0 / 29.0)) * _D65_XN
 424      yn = np.where(fy > delta, fy ** 3, 3.0 * delta * delta * (fy - 4.0 / 29.0)) * _D65_YN
 425      zn = np.where(fz > delta, fz ** 3, 3.0 * delta * delta * (fz - 4.0 / 29.0)) * _D65_ZN
 426  
 427      # XYZ -> linear RGB (same matrix as scalar version)
 428      rl = 3.2404542 * xn - 1.5371385 * yn - 0.4985314 * zn
 429      gl = -0.9692660 * xn + 1.8760108 * yn + 0.0415560 * zn
 430      bl = 0.0556434 * xn - 0.2040259 * yn + 1.0572252 * zn
 431  
 432      # Gamma correction (same thresholds as scalar version)
 433      rl = np.maximum(0.0, rl)
 434      gl = np.maximum(0.0, gl)
 435      bl = np.maximum(0.0, bl)
 436  
 437      rl = np.where(rl > 0.0031308, 1.055 * rl ** (1.0 / 2.4) - 0.055, 12.92 * rl)
 438      gl = np.where(gl > 0.0031308, 1.055 * gl ** (1.0 / 2.4) - 0.055, 12.92 * gl)
 439      bl = np.where(bl > 0.0031308, 1.055 * bl ** (1.0 / 2.4) - 0.055, 12.92 * bl)
 440  
 441      rgb = np.column_stack([
 442          np.clip(np.round(rl * 255), 0, 255),
 443          np.clip(np.round(gl * 255), 0, 255),
 444          np.clip(np.round(bl * 255), 0, 255),
 445      ]).astype(np.int32)
 446  
 447      return rgb

 398  def _extract_face_landmarks(image: Image.Image) -> Optional[Dict[str, np.ndarray]]:
 399      """Extract face mesh landmarks using MediaPipe Face Mesh."""
 400      try:
 401          import mediapipe as mp
 402          img_array = np.array(image.convert("RGB"))
 403          h, w = img_array.shape[:2]
 404  
 405          with mp.solutions.face_mesh.FaceMesh(
 406              static_image_mode=True,
 407              max_num_faces=1,
 408              refine_landmarks=True,
 409              min_detection_confidence=0.5,
 410          ) as face_mesh:
 411              results = face_mesh.process(img_array)
 412              if not results.multi_face_landmarks:
 413                  return None
 414  
 415              landmarks = results.multi_face_landmarks[0]
 416              regions = {}
 417              for region_name, indices in _FACE_MESH_REGIONS.items():
 418                  points = []
 419                  for idx in indices:
 420                      if idx < len(landmarks.landmark):
 421                          lm = landmarks.landmark[idx]
 422                          points.append((lm.x * w, lm.y * h))
 423                  if points:
 424                      regions[region_name] = np.array(points)
 425  
 426              return regions
 427      except ImportError:
 428          logger.debug("MediaPipe Face Mesh not available, using fallback regions")
 429          return None
 430      except Exception as e:
 431          logger.debug("Face mesh extraction failed: %s", str(e))
 432          return None

 435  def _extract_region_pixels(
 436      img_array: np.ndarray, region: Tuple[float, float, float, float]
 437  ) -> np.ndarray:
 438      """Extract pixels from a normalized region (x_ratio, y_ratio, w_ratio, h_ratio)."""
 439      h, w = img_array.shape[:2]
 440      x1 = int(region[0] * w)
 441      y1 = int(region[1] * h)
 442      x2 = int((region[0] + region[2]) * w)
 443      y2 = int((region[1] + region[3]) * h)
 444      x1, x2 = max(0, x1), min(w, x2)
 445      y1, y2 = max(0, y1), min(h, y2)
 446      return img_array[y1:y2, x1:x2]

 448  
 449  def _compute_region_lab(
 450      img_array: np.ndarray,
 451      region_coords: Optional[Tuple[float, float, float, float]] = None,
 452      landmark_points: Optional[np.ndarray] = None,
 453  ) -> Optional[Tuple[float, float, float]]:
 454      """Compute median CIELAB for skin pixels in a region."""
 455      if landmark_points is not None and len(landmark_points) >= 3:
 456          # Use landmark convex hull
 457          import cv2
 458          hull = cv2.convexHull(landmark_points.astype(np.int32))
 459          mask = np.zeros(img_array.shape[:2], dtype=np.uint8)
 460          cv2.fillConvexPoly(mask, hull, 255)
 461          pixels = img_array[mask > 0]
 462      elif region_coords is not None:
 463          pixels = _extract_region_pixels(img_array, region_coords)
 464          if pixels.size == 0:
 465              return None
 466          pixels = pixels.reshape(-1, 3)
 467      else:
 468          return None
 469  
 470      lab_l = []
 471      lab_a = []
 472      lab_b = []
 473  
 474      for pixel in pixels:
 475          r, g, b = int(pixel[0]), int(pixel[1]), int(pixel[2])
 476          if _is_skin_pixel_cielab(r, g, b):
 477              l_val, a_val, b_val = _rgb_to_lab(r, g, b)
 478              lab_l.append(l_val)
 479              lab_a.append(a_val)
 480              lab_b.append(b_val)
 481  
 482      if len(lab_l) < 5:
 483          return None
 484  
 485      return (
 486          float(np.median(lab_l)),
 487          float(np.median(lab_a)),
 488          float(np.median(lab_b)),
 489      )

 493  # Season Classification Algorithm

 496  WARM_B_THRESHOLD = 15.0
 497  
 498  
 499  def _classify_tone(a_star: float, b_star: float = 15.0) -> Tuple[ToneType, float]:
 500      """Classify warm/cool tone from CIELAB a* and b* axes.
 501  
 502      In CIELAB:
 503      - positive a* = red/warm, negative a* = green/cool
 504      - positive b* = yellow/warm, negative b* = blue/cool
 505  
 506      For Asian skin tones, b* (yellowness) is a critical dimension
 507      for warm/cool classification. b* > 15 indicates warm undertone.
 508      The final classification combines both axes with weighted scoring.
 509      """
 510      a_warm_score = 0.0
 511      if a_star > 8.0:
 512          a_warm_score = min(1.0, 0.5 + a_star / 30.0)
 513      elif a_star > 3.0:
 514          a_warm_score = 0.5 + a_star / 20.0
 515      elif a_star < -2.0:
 516          a_warm_score = max(0.0, 0.5 - abs(a_star) / 20.0)
 517      else:
 518          a_warm_score = 0.5
 519  
 520      b_warm_score = 0.0
 521      if b_star > WARM_B_THRESHOLD:
 522          b_warm_score = min(1.0, 0.5 + (b_star - WARM_B_THRESHOLD) / 20.0)
 523      elif b_star > 10.0:
 524          b_warm_score = 0.5 + (b_star - 10.0) / 10.0
 525      elif b_star < 5.0:
 526          b_warm_score = max(0.0, 0.5 - (5.0 - b_star) / 10.0)
 527      else:
 528          b_warm_score = 0.5
 529  
 530      combined = 0.4 * a_warm_score + 0.6 * b_warm_score
 531  
 532      if combined >= 0.6:
 533          return ToneType.WARM, min(1.0, combined)
 534      if combined <= 0.4:
 535          return ToneType.COOL, min(1.0, 1.0 - combined)
 536      return ToneType.NEUTRAL, 0.5
 537  
 538  
 539  def _classify_depth(l_star: float) -> Tuple[DepthType, float]:
 540      """Classify light/deep from CIELAB L* axis.
 541  
 542      Threshold adjusted to 58 for Chinese population whose skin tones
 543      are generally darker than Caucasian baselines. L*=65 caused
 544      systematic misclassification of medium-deep skin as "light".
 545      """
 546      if l_star >= 58.0:
 547          return DepthType.LIGHT, min(1.0, 0.5 + (l_star - 58.0) / 30.0)
 548      if l_star >= 45.0:
 549          return DepthType.LIGHT, 0.5 + (l_star - 45.0) / 26.0
 550      if l_star >= 35.0:
 551          return DepthType.DEEP, 0.5 + (45.0 - l_star) / 20.0
 552      return DepthType.DEEP, min(1.0, 0.5 + (35.0 - l_star) / 20.0)
 553  
 554  
 555  def _classify_chroma(chroma: float) -> Tuple[ChromaType, float]:
 556      """Classify clear/muted from CIELAB chroma C*."""
 557      if chroma >= 18.0:
 558          return ChromaType.CLEAR, min(1.0, 0.5 + (chroma - 18.0) / 15.0)
 559      if chroma >= 12.0:
 560          return ChromaType.CLEAR, 0.5 + (chroma - 12.0) / 12.0
 561      if chroma >= 8.0:
 562          return ChromaType.MUTED, 0.5 + (12.0 - chroma) / 8.0
 563      return ChromaType.MUTED, min(1.0, 0.5 + (8.0 - chroma) / 8.0)
 564  
 565  
 566  def _determine_season(
 567      tone: ToneType, depth: DepthType, chroma: ChromaType
 568  ) -> TwelveSeason:
 569      """Determine 12-season type from tone, depth, and chroma."""
 570      if tone in (ToneType.WARM, ToneType.NEUTRAL):
 571          if depth == DepthType.LIGHT:
 572              if chroma == ChromaType.CLEAR:
 573                  return TwelveSeason.SPRING_WARM_LIGHT_CLEAR
 574              return TwelveSeason.SPRING_WARM_LIGHT_MUTED
 575          # Deep + Warm
 576          if chroma == ChromaType.CLEAR:
 577              return TwelveSeason.SPRING_WARM_DEEP_CLEAR
 578          return TwelveSeason.AUTUMN_WARM_DEEP_MUTED
 579      else:
 580          # Cool
 581          if depth == DepthType.LIGHT:
 582              if chroma == ChromaType.CLEAR:
 583                  return TwelveSeason.SUMMER_COOL_LIGHT_CLEAR
 584              return TwelveSeason.SUMMER_COOL_LIGHT_MUTED
 585          # Deep + Cool
 586          if chroma == ChromaType.CLEAR:
 587              return TwelveSeason.WINTER_COOL_DEEP_CLEAR
 588          return TwelveSeason.WINTER_COOL_DEEP_MUTED

 595  def generate_palette(season: TwelveSeason) -> Dict[str, List[ColorSwatch]]:
 596      """Generate color palette for a 12-season type."""
 597      palette_data = _SEASON_PALETTES.get(season)
 598      if palette_data is None:
 599          logger.warning("Unknown season type: %s, falling back to SPRING_WARM_LIGHT_CLEAR", season)
 600          palette_data = _SEASON_PALETTES[TwelveSeason.SPRING_WARM_LIGHT_CLEAR]
 601  
 602      suitable = [
 603          ColorSwatch(name=c["name"], hex_value=c["hex"], reason=c["reason"])
 604          for c in palette_data["suitable"]
 605      ]
 606      unsuitable = [
 607          ColorSwatch(name=c["name"], hex_value=c["hex"], reason=c["reason"])
 608          for c in palette_data["unsuitable"]
 609      ]
 610  
 611      return {"suitable": suitable, "unsuitable": unsuitable}

 615  # Main Analysis Entry Point
 616  # ============================================================
 617  
 618  def analyze_color_season(
 619      face_image: Union[str, Image.Image],
 620  ) -> ColorSeasonResult:
 621      """Analyze color season from a face image using CIELAB + ITA + 12-season system.
 622  
 623      Pipeline:
 624      1. Extract face mesh landmarks (MediaPipe Face Mesh 468 points)
 625      2. Sample skin pixels from facial regions using landmark convex hulls
 626      3. Compute average CIELAB values (L*, a*, b*)
 627      4. Compute ITA (Individual Typology Angle)
 628      5. Classify along 3 dimensions: hue (a*), value (L*), chroma (C*)
 629      6. Map to one of 12 seasons
 630      7. Generate color palette with CIEDE2000 harmony verification
 631  
 632      Args:
 633          face_image: Path to image file or PIL Image object of a face.
 634  
 635      Returns:
 636          ColorSeasonResult with 12-season type, tone, depth, chroma,
 637          skin CIELAB values, ITA score, and color palette.
 638      """
 639      if isinstance(face_image, str):
 640          try:
 641              img = Image.open(face_image).convert("RGB")
 642          except Exception as e:
 643              logger.error("Failed to open image %s: %s", face_image, e)
 644              raise ValueError(f"Cannot open image: {face_image}") from e
 645      elif isinstance(face_image, Image.Image):
 646          img = face_image.convert("RGB")
 647      else:
 648          raise TypeError(f"Expected str or PIL.Image, got {type(face_image)}")
 649  
 650      img_array = np.array(img)
 651      logger.info(
 652          "Analyzing color season from image of size %sx%s",
 653          img_array.shape[1], img_array.shape[0],
 654      )
 655  
 656      # Step 1: Try MediaPipe Face Mesh for precise regions
 657      face_regions = _extract_face_landmarks(img)
 658  
 659      # Step 2: Extract CIELAB values from facial regions
 660      region_results: Dict[str, Tuple[float, float, float]] = {}
 661  
 662      if face_regions:
 663          # Use landmark-based regions
 664          for region_name, points in face_regions.items():
 665              result = _compute_region_lab(img_array, landmark_points=points)
 666              if result is not None:
 667                  region_results[region_name] = result
 668                  logger.debug(
 669                      "Region %s (landmark): L=%.1f a=%.1f b=%.1f",
 670                      region_name, result[0], result[1], result[2],
 671                  )
 672  
 673      # Fallback to fixed regions if landmark extraction failed
 674      if not face_regions:
 675          for region_name, region_coords in _FALLBACK_REGIONS.items():
 676              result = _compute_region_lab(img_array, region_coords=region_coords)
 677              if result is not None:
 678                  region_results[region_name] = result
 679                  logger.debug(
 680                      "Region %s (fallback): L=%.1f a=%.1f b=%.1f",
 681                      region_name, result[0], result[1], result[2],
 682                  )
 683  
 684      if not region_results:
 685          logger.warning("No valid skin regions detected, using fallback defaults")
 686          avg_l, avg_a, avg_b = 65.0, 8.0, 15.0
 687      else:
 688          # Weighted average of regions
 689          weights: Dict[str, float] = {
 690              "forehead": 0.25,
 691              "left_cheek": 0.20,
 692              "right_cheek": 0.20,
 693              "nose_bridge": 0.15,
 694              "chin": 0.20,
 695          }
 696          total_weight = sum(weights.get(name, 0.1) for name in region_results)
 697          avg_l = sum(
 698              region_results[n][0] * weights.get(n, 0.1) for n in region_results
 699          ) / total_weight
 700          avg_a = sum(
 701              region_results[n][1] * weights.get(n, 0.1) for n in region_results
 702          ) / total_weight
 703          avg_b = sum(
 704              region_results[n][2] * weights.get(n, 0.1) for n in region_results
 705          ) / total_weight
 706  
 707      logger.info(
 708          "Weighted average skin CIELAB: L=%.1f a=%.1f b=%.1f",
 709          avg_l, avg_a, avg_b,
 710      )
 711  
 712      # Step 3: Compute derived metrics
 713      chroma = _compute_chroma(avg_a, avg_b)
 714      ita = _compute_ita(avg_l, avg_b)
 715  
 716      logger.info("Chroma=%.1f, ITA=%.1f", chroma, ita)
 717  
 718      # Step 4: Classify along 3 dimensions
 719      tone, tone_conf = _classify_tone(avg_a, avg_b)
 720      depth, depth_conf = _classify_depth(avg_l)
 721      chroma_type, chroma_conf = _classify_chroma(chroma)
 722  
 723      # Step 5: Map to 12-season
 724      season = _determine_season(tone, depth, chroma_type)
 725  
 726      # Overall confidence
 727      confidence = (tone_conf + depth_conf + chroma_conf) / 3.0
 728  
 729      logger.info(
 730          "Color season result: %s (tone=%s, depth=%s, chroma=%s, ITA=%.1f, confidence=%.3f)",
 731          season.value, tone.value, depth.value, chroma_type.value, ita, confidence,
 732      )
 733  
 734      # Step 6: Generate palette
 735      palette = generate_palette(season)
 736  
 737      return ColorSeasonResult(
 738          season=season,
 739          parent_season=_PARENT_SEASON[season],
 740          tone=tone,
 741          depth=depth,
 742          chroma=chroma_type,
 743          skin_lab=(avg_l, avg_a, avg_b),
 744          ita=ita,
 745          suitable_colors=palette["suitable"],
 746          unsuitable_colors=palette["unsuitable"],
 747          confidence=confidence,
 748      )

 767  class EightSeason(Enum):
 768      """8-season color analysis system aligned with Prisma schema.
 769  
 770      The 8-season system combines the 3-dimension classification
 771      (hue: warm/cool, value: light/deep) into a practical 8-type model:
 772      - Each base season (spring/summer/autumn/winter) has two sub-types:
 773        one emphasizing the dominant characteristic and one emphasizing
 774        the secondary characteristic (light for spring/summer, deep for autumn/winter).
 775      """
 776      SPRING_WARM = "spring_warm"    # Warm Spring: warm + light, dominant warm
 777      SPRING_LIGHT = "spring_light"  # Light Spring: warm + light, dominant light
 778      SUMMER_COOL = "summer_cool"    # Cool Summer: cool + light, dominant cool
 779      SUMMER_LIGHT = "summer_light"  # Light Summer: cool + light, dominant light
 780      AUTUMN_WARM = "autumn_warm"    # Warm Autumn: warm + deep, dominant warm
 781      AUTUMN_DEEP = "autumn_deep"    # Deep Autumn: warm + deep, dominant deep
 782      WINTER_COOL = "winter_cool"    # Cool Winter: cool + deep, dominant cool
 783      WINTER_DEEP = "winter_deep"    # Deep Winter: cool + deep, dominant deep
