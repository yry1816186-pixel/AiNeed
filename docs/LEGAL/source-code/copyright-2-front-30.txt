  90  export default function App() {
  91    const {
  92      isAuthenticated,
  93      isLoading,
  94      REDACTED,
  95      setLoading,
  96    } = useAuthStore(
  97      useShallow((state) => ({
  98        isAuthenticated: state.isAuthenticated,
  99        isLoading: state.isLoading,
 100        REDACTEDstate.accessToken,
 101        setLoading: state.setLoading,
 102      }))
 103    );
 104  
 105    const [currentRouteName, setCurrentRouteName] = useState<string | undefined>();
 106  
 107    const pendingDeepLinkUrlRef = useRef<string | null>(null);
 108    const lastHandledDeepLinkUrlRef = useRef<string | null>(null);
 109    const hasAutoLoggedInRef = useRef(false);
 110  
 111    const queueDeepLink = useCallback((url?: string | null) => {
 112      if (!url) {
 113        return;
 114      }
 115      pendingDeepLinkUrlRef.current = url;
 116    }, []);
 117  
 118    const flushPendingDeepLink = useCallback(() => {
 119      if (!checkNavigationReady() || isLoading) {
 120        return;
 121      }
 122  
 123      const pendingUrl = pendingDeepLinkUrlRef.current;
 124      if (!pendingUrl || pendingUrl === lastHandledDeepLinkUrlRef.current) {
 125        return;
 126      }
 127  
 128      const handled = navigateDeepLink(pendingUrl, isAuthenticated);
 129      if (handled) {
 130        lastHandledDeepLinkUrlRef.current = pendingUrl;
 131        pendingDeepLinkUrlRef.current = null;
 132      }
 133    }, [isAuthenticated, isLoading]);
 134  
 135    useEffect(() => {
 136      if (
 137        DEV_TEST_ACCOUNT_CONFIG.enabled &&
 138        !isLoading &&
 139        !isAuthenticated &&
 140        !authToken &&
 141        !hasAutoLoggedInRef.current
 142      ) {
 143        hasAutoLoggedInRef.current = true;
 144        const autoLogin = async () => {
 145          try {
 146            console.log("[DEV] Attempting auto-login with test account...");
 147            const response = await authApi.login({
 148              email: DEV_TEST_ACCOUNT_CONFIG.email,
 149              REDACTEDST_ACCOUNT_CONFIG.password,
 150            });
 151  
 152            if (response.success && response.data) {
 153              useAuthStore.getState().setToken(response.data.token);
 154              useAuthStore.getState().setUser(response.data.user);
 155              console.log("[DEV] Auto-login successful");
 156            }
 157          } catch (error) {
 158            console.warn("[DEV] Auto-login failed:", error);
 159            hasAutoLoggedInRef.current = false;
 160          }
 161        };
 162  
 163        const timer = setTimeout(autoLogin, 500);
 164        return () => clearTimeout(timer);
 165      }
 166  
 167      return undefined;
 168    }, [authToken, isAuthenticated, isLoading]);
 169  
 170    useEffect(() => {
 171      apiClient.onAuthExpired(() => {
 172        useAuthStore.getState().logout();
 173      });
 174      analytics.init();
 175  
 176      return () => {
 177        analytics.destroy();
 178      };
 179    }, []);
 180  
 181    useEffect(() => {
 182      const unsub = useAuthStore.persist.onFinishHydration(() => {
 183        setLoading(false);
 184      });
 185  
 186      if (useAuthStore.persist.hasHydrated()) {
 187        setLoading(false);
 188      }
 189  
 190      const timeout = setTimeout(() => {
 191        setLoading(false);
 192      }, 3000);
 193  
 194      return () => {
 195        unsub();
 196        clearTimeout(timeout);
 197      };
 198    }, [setLoading]);
 199  
 200    useEffect(() => {
 201      if (isLoading) {
 202        return;
 203      }
 204      void apiClient.setToken(authToken ?? null);
 205    }, [authToken, isLoading]);
 206  
 207    useEffect(() => {
 208      let isMounted = true;
 209  
 210      void Linking.getInitialURL().then((url) => {
 211        if (isMounted) {
 212          queueDeepLink(url);
 213        }
 214      });
 215  
 216      const subscription = Linking.addEventListener("url", ({ url }) => {
 217        queueDeepLink(url);
 218      });
 219  
 220      return () => {
 221        isMounted = false;
 222        subscription.remove();
 223      };
 224    }, [queueDeepLink]);
 225  
 226    useEffect(() => {
 227      flushPendingDeepLink();
 228    }, [currentRouteName, flushPendingDeepLink, isAuthenticated]);
 229  
 230    const navigationReadyHandler = useMemo(
 231      () => ({
 232        onReady: () => {
 233          setNavigationReady(true);
 234          const routeName = navigationRef.getCurrentRoute()?.name;
 235          setCurrentRouteName(routeName);
 236          if (routeName) {
 237            analytics.trackScreen(routeName);
 238          }
 239        },
 240        onStateChange: () => {
 241          const routeName = navigationRef.getCurrentRoute()?.name;
 242          setCurrentRouteName(routeName);
 243          if (routeName) {
 244            analytics.trackScreen(routeName);
 245          }
 246        },
 247      }),
 248      []
 249    );
 250  
 251    if (isLoading) {
 252      return (
 253        <ErrorBoundary
 254          screenName="SplashScreen"
 255          context={{ phase: "loading" }}
 256          onError={(error, errorInfo, structuredError) => {
 257            console.error("[App:Loading] Error:", structuredError);
 258          }}
 259        >
 260          <AppProviders>
 261            <ThemedStatusBar />
 262            <SplashScreen />
 263          </AppProviders>
 264        </ErrorBoundary>
 265      );
 266    }
 267  
 268    return (
 269      <ErrorBoundary
 270        screenName="AppRoot"
 271        context={{ isAuthenticated }}
 272        maxRetries={3}
 273        onError={(error, errorInfo, structuredError) => {
 274          console.error("[App:Root] Error:", structuredError);
 275        }}
 276        onReset={() => {
 277          console.log("[App:Root] Error boundary reset");
 278        }}
 279      >
 280        <AppProviders>
 281          <NavigationContainer
 282            ref={navigationRef}
 283            onReady={navigationReadyHandler.onReady}
 284            onStateChange={navigationReadyHandler.onStateChange}
 285          >
 286            <ThemedStatusBar />
 287            <OfflineBanner />
 288            <RootNavigator isAuthenticated={isAuthenticated} />
 289          </NavigationContainer>
 290        </AppProviders>
 291      </ErrorBoundary>
 292    );
 293  }
