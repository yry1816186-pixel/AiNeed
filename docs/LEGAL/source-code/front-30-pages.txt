  48  class ResponseCache:
  49      """LRU cache for LLM responses with TTL support"""
  50      
  51      def __init__(self, max_size: int = 500, ttl_seconds: int = 1800):
  52          self.max_size = max_size
  53          self.ttl_seconds = ttl_seconds
  54          self._cache: OrderedDict[str, tuple] = OrderedDict()
  55          self._lock = threading.RLock()
  56          self._hits = 0
  57          self._misses = 0
  58      
  59      def _compute_key(self, *args, **kwargs) -> str:
  60          """Compute cache key from arguments"""
  61          key_data = json.dumps({"args": args, "kwargs": kwargs}, sort_keys=True, default=str)
  62          return hashlib.sha256(key_data.encode('utf-8')).hexdigest()
  63      
  64      def get(self, key: str) -> Optional[Any]:
  65          """Get cached response if exists and not expired"""
  66          with self._lock:
  67              if key not in self._cache:
  68                  self._misses += 1
  69                  return None
  70              
  71              value, timestamp = self._cache[key]
  72              
  73              if time.time() - timestamp > self.ttl_seconds:
  74                  del self._cache[key]
  75                  self._misses += 1
  76                  return None
  77              
  78              self._cache.move_to_end(key)
  79              self._hits += 1
  80              return value
  81      
  82      def set(self, key: str, value: Any) -> None:
  83          """Store response in cache"""
  84          with self._lock:
  85              if key in self._cache:
  86                  del self._cache[key]
  87              
  88              self._cache[key] = (value, time.time())
  89              
  90              while len(self._cache) > self.max_size:
  91                  self._cache.popitem(last=False)
  92      
  93      def invalidate_pattern(self, pattern: str) -> int:
  94          """Invalidate cache entries matching pattern prefix"""
  95          with self._lock:
  96              keys_to_delete = [k for k in self._cache.keys() if k.startswith(pattern)]
  97              for key in keys_to_delete:
  98                  del self._cache[key]
  99              return len(keys_to_delete)
 100      
 101      def clear(self) -> None:
 102          """Clear all cached entries"""
 103          with self._lock:
 104              self._cache.clear()
 105      
 106      def get_stats(self) -> Dict[str, Any]:
 107          """Get cache statistics"""
 108          with self._lock:
 109              total = self._hits + self._misses
 110              hit_rate = self._hits / total if total > 0 else 0
 111              return {
 112                  "size": len(self._cache),
 113                  "max_size": self.max_size,
 114                  "hits": self._hits,
 115                  "misses": self._misses,
 116                  "hit_rate": round(hit_rate, 4),
 117                  "ttl_seconds": self.ttl_seconds
 118              }

 127  class CircuitBreaker:
 128      """Circuit breaker for LLM API calls"""
 129      
 130      def __init__(
 131          self,
 132          failure_threshold: int = 5,
 133          recovery_timeout: int = 60,
 134          half_open_max_calls: int = 3
 135      ):
 136          self.failure_threshold = failure_threshold
 137          self.recovery_timeout = recovery_timeout
 138          self.half_open_max_calls = half_open_max_calls
 139          self.failure_count = 0
 140          self.last_failure_time: Optional[float] = None
 141          self.state = CircuitState.CLOSED
 142          self.half_open_calls = 0
 143      
 144      def can_execute(self) -> bool:
 145          if self.state == CircuitState.CLOSED:
 146              return True
 147          if self.state == CircuitState.OPEN:
 148              if self.last_failure_time and time.time() - self.last_failure_time >= self.recovery_timeout:
 149                  self.state = CircuitState.HALF_OPEN
 150                  self.half_open_calls = 0
 151                  logger.info("Circuit breaker entering HALF_OPEN state")
 152                  return True
 153              return False
 154          if self.state == CircuitState.HALF_OPEN:
 155              if self.half_open_calls < self.half_open_max_calls:
 156                  self.half_open_calls += 1
 157                  return True
 158              return False
 159          return False
 160      
 161      def record_success(self):
 162          if self.state == CircuitState.HALF_OPEN:
 163              self.state = CircuitState.CLOSED
 164              self.failure_count = 0
 165              logger.info("Circuit breaker recovered to CLOSED state")
 166      
 167      def record_failure(self):
 168          self.failure_count += 1
 169          self.last_failure_time = time.time()
 170          if self.state == CircuitState.HALF_OPEN:
 171              self.state = CircuitState.OPEN
 172              logger.warning("Circuit breaker returned to OPEN state")
 173          elif self.failure_count >= self.failure_threshold:
 174              self.state = CircuitState.OPEN
 175              logger.error(f"Circuit breaker opened after {self.failure_count} failures")

 188  def with_retry_and_circuit_breaker(
 189      max_retries: int = 3,
 190      base_delay: float = 1.0,
 191      max_delay: float = 30.0,
 192      exponential_base: float = 2.0,
 193      circuit_breaker: Optional[CircuitBreaker] = None
 194  ):
 195      """Decorator for retry with exponential backoff and circuit breaker
 196  
 197      A-P2-1: 增强重试逻辑
 198      - 区分可重试错误（RetryableError）和不可重试错误（NonRetryableError）
 199      - 可重试错误触发指数退避重试
 200      - 不可重试错误立即抛出，不重试
 201      - 熔断器仅对可重试错误计数
 202      """
 203  
 204      def decorator(func):
 205          @wraps(func)
 206          async def wrapper(*args, **kwargs):
 207              if circuit_breaker and not circuit_breaker.can_execute():
 208                  raise NonRetryableError("Circuit breaker is OPEN - LLM service unavailable")
 209  
 210              last_exception = None
 211              for attempt in range(max_retries):
 212                  try:
 213                      result = await func(*args, **kwargs)
 214                      if circuit_breaker:
 215                          circuit_breaker.record_success()
 216                      return result
 217                  except NonRetryableError as e:
 218                      # 不可重试错误：立即抛出
 219                      logger.error(f"Non-retryable error, aborting: {e}")
 220                      raise
 221                  except (RetryableError, Exception) as e:
 222                      last_exception = e
 223                      if circuit_breaker:
 224                          circuit_breaker.record_failure()
 225  
 226                      if attempt < max_retries - 1:
 227                          delay = min(base_delay * (exponential_base ** attempt), max_delay)
 228                          # 如果是 429 错误，尝试从异常信息中提取 Retry-After
 229                          retry_after = getattr(e, 'retry_after', None)
 230                          if retry_after and retry_after > 0:
 231                              delay = max(delay, min(retry_after, max_delay))
 232                          logger.warning(
 233                              f"LLM API call failed (attempt {attempt + 1}/{max_retries}): {e}. "
 234                              f"Retrying in {delay:.1f}s..."
 235                          )
 236                          await asyncio.sleep(delay)
 237                      else:
 238                          logger.error(f"LLM API call failed after {max_retries} attempts: {e}")
 239  
 240              raise last_exception
 241          return wrapper
 242      return decorator

 330      def _count_tokens(self, text: str) -> int:
 331          """
 332          A-P1-7: Improved token counting with tiktoken fallback.
 333  
 334          Token counting strategy (in order of preference):
 335          1. tiktoken library (if available) - most accurate for GPT-compatible tokenizers
 336          2. Character-based estimation with improved accuracy for mixed CJK/Latin text
 337  
 338          Estimation rules for fallback:
 339          - CJK characters: ~1.5 tokens/char (refined from 1.8 based on empirical data)
 340          - Full-width punctuation/kana: ~1.2 tokens/char
 341          - English words: ~1.3 tokens/word (average across word lengths)
 342          - Punctuation and whitespace: ~0.3 tokens/char
 343          - Numbers: ~0.5 tokens/digit group
 344          """
 345          if not text:
 346              return 0
 347  
 348          # Strategy 1: Try tiktoken for accurate counting
 349          try:
 350              import tiktoken
 351              # Use cl100k_base encoding (GPT-4/GLM compatible)
 352              enc = tiktoken.get_encoding("cl100k_base")
 353              return len(enc.encode(text))
 354          except ImportError:
 355              pass
 356          except Exception:
 357              pass
 358  
 359          # Strategy 2: Improved character-based estimation
 360          token_count = 0.0
 361          i = 0
 362          while i < len(text):
 363              char = text[i]
 364  
 365              # CJK Unified Ideographs (common Chinese characters)
 366              if '\u4e00' <= char <= '\u9fff':
 367                  token_count += 1.5
 368              # CJK Extension A and B (rare characters)
 369              elif '\u3400' <= char <= '\u4dbf' or '\U00020000' <= char <= '\U0002a6df':
 370                  token_count += 1.5
 371              # Full-width punctuation, CJK symbols
 372              elif '\u3000' <= char <= '\u303f':
 373                  token_count += 1.2
 374              # Hiragana and Katakana
 375              elif '\u3040' <= char <= '\u309f' or '\u30a0' <= char <= '\u30ff':
 376                  token_count += 1.2
 377              # Hangul syllables
 378              elif '\uac00' <= char <= '\ud7af':
 379                  token_count += 1.5
 380              # English letters and numbers - process as word groups
 381              elif char.isascii() and char.isalnum():
 382                  word_start = i
 383                  while i < len(text) and text[i].isascii() and text[i].isalnum():
 384                      i += 1
 385                  word_length = i - word_start
 386                  # English word token ratio: shorter words have higher ratio
 387                  # 1 char ~1 token, 2 chars ~1.5, 3+ chars ~1.3/word
 388                  if word_length <= 1:
 389                      token_count += 1.0
 390                  elif word_length <= 2:
 391                      token_count += 1.5
 392                  else:
 393                      token_count += max(1.0, word_length * 0.4)
 394                  continue  # Skip i increment, already advanced
 395              # ASCII punctuation and whitespace
 396              elif char.isascii():
 397                  token_count += 0.3
 398              # Other Unicode (emoji, etc.)
 399              else:
 400                  token_count += 1.5
 401  
 402              i += 1
 403  
 404          return max(1, int(token_count))

 406      def _truncate_prompt(
 407          self,
 408          system_prompt: str,
 409          user_prompt: str,
 410          max_output_tokens: int = 4096
 411      ) -> tuple:
 412          """
 413          P1-006修复: Prompt截断逻辑
 414  
 415          当输入token超过限制时，智能截断用户prompt，保留系统prompt完整性。
 416  
 417          截断策略：
 418          1. 始终保留系统prompt（包含核心指令）
 419          2. 优先截断用户prompt的中部和尾部
 420          3. 保留用户prompt的开头和结尾关键信息
 421          4. 添加截断标记提示
 422  
 423          Args:
 424              system_prompt: 系统提示词
 425              user_prompt: 用户提示词
 426              max_output_tokens: 最大输出token数
 427  
 428          Returns:
 429              tuple: (截断后的system_prompt, 截断后的user_prompt, 是否被截断)
 430          """
 431          system_tokens = self._count_tokens(system_prompt)
 432          user_tokens = self._count_tokens(user_prompt)
 433  
 434          # 计算可用的输入token空间
 435          available_input_tokens = (
 436              self.MAX_CONTEXT_TOKENS
 437              - max_output_tokens
 438              - self.SAFETY_MARGIN
 439              - system_tokens
 440          )
 441  
 442          # 如果用户prompt在限制内，无需截断
 443          if user_tokens <= available_input_tokens:
 444              return system_prompt, user_prompt, False
 445  
 446          logger.warning(
 447              f"Prompt truncation required: user_tokens={user_tokens}, "
 448              f"available={available_input_tokens}"
 449          )
 450  
 451          # 执行截断
 452          self._truncation_counter += 1
 453  
 454          # 计算需要保留的字符比例
 455          target_ratio = available_input_tokens / user_tokens
 456  
 457          # 截断策略：保留开头60%和结尾40%的目标内容
 458          # 这样可以保留问题的背景和具体请求
 459          keep_ratio_head = 0.6 * target_ratio
 460          keep_ratio_tail = 0.4 * target_ratio
 461  
 462          truncate_point_head = int(len(user_prompt) * keep_ratio_head)
 463          truncate_point_tail = int(len(user_prompt) * (1 - keep_ratio_tail))
 464  
 465          # 在句子边界截断（寻找标点符号）
 466          # 向前寻找最近的句子结束点
 467          for offset in range(min(100, truncate_point_head)):
 468              idx = truncate_point_head - offset
 469              if idx > 0 and user_prompt[idx] in '。！？\n':
 470                  truncate_point_head = idx + 1
 471                  break
 472  
 473          # 向后寻找最近的句子开始点
 474          for offset in range(min(100, len(user_prompt) - truncate_point_tail)):
 475              idx = truncate_point_tail + offset
 476              if idx < len(user_prompt) and user_prompt[idx] in '。！？\n':
 477                  truncate_point_tail = idx + 1
 478                  break
 479  
 480          # 构建截断后的prompt
 481          truncation_marker = "\n\n[...内容已省略...]\n\n"
 482          truncated_user_prompt = (
 483              user_prompt[:truncate_point_head]
 484              + truncation_marker
 485              + user_prompt[truncate_point_tail:]
 486          )
 487  
 488          # 验证截断后的token数
 489          new_user_tokens = self._count_tokens(truncated_user_prompt)
 490          if new_user_tokens > available_input_tokens:
 491              # 如果仍然超限，执行更激进的截断
 492              # 只保留开头部分
 493              safe_length = int(len(user_prompt) * target_ratio * 0.8)
 494              truncated_user_prompt = (
 495                  user_prompt[:safe_length]
 496                  + "\n\n[...内容因长度限制已省略，请根据以上信息提供建议...]"
 497              )
 498  
 499          logger.info(
 500              f"Prompt truncated: original={user_tokens} tokens, "
 501              f"truncated={self._count_tokens(truncated_user_prompt)} tokens"
 502          )
 503  
 504          return system_prompt, truncated_user_prompt, True

 519      @with_retry_and_circuit_breaker(max_retries=3, base_delay=1.0)
 520      async def generate(self, system_prompt: str, user_prompt: str, max_tokens: int = 4000) -> str:
 521          # P1-006修复: 执行prompt截断检查
 522          system_prompt, user_prompt, was_truncated = self._truncate_prompt(
 523              system_prompt, user_prompt, max_tokens
 524          )
 525  
 526          if was_truncated:
 527              logger.warning("Prompt was truncated to fit within token limits")
 528  
 529          estimated_tokens = self._count_tokens(system_prompt) + self._count_tokens(user_prompt) + max_tokens
 530  
 531          if not await rate_limiter.wait_and_acquire("glm", estimated_tokens, max_wait_ms=30000):
 532              raise Exception("Rate limit exceeded - please try again later")
 533  
 534          headers = {
 535              "Authorization": f"Bearer {self.api_key}",
 536              "Content-Type": "application/json"
 537          }
 538  
 539          payload = {
 540              "model": self.model,
 541              "messages": [
 542                  {"role": "system", "content": system_prompt},
 543                  {"role": "user", "content": user_prompt}
 544              ],
 545              "max_tokens": max_tokens,
 546              "temperature": 0.8,
 547              "top_p": 0.9
 548          }
 549  
 550          self._request_counter += 1
 551          self._token_counter += self._count_tokens(system_prompt) + self._count_tokens(user_prompt)
 552  
 553          try:
 554              async with aiohttp.ClientSession() as session:
 555                  async with session.post(
 556                      f"{self.api_endpoint}/chat/completions",
 557                      headers=headers,
 558                      json=payload,
 559                      timeout=aiohttp.ClientTimeout(total=180)
 560                  ) as response:
 561                      if response.status == 200:
 562                          result = await response.json()
 563                          content = result.get("choices", [{}])[0].get("message", {}).get("content", "")
 564                          self._token_counter += self._count_tokens(content)
 565                          return content
 566                      elif response.status == 429:
 567                          # A-P2-1: 429 为可重试错误，提取 Retry-After
 568                          retry_after_header = response.headers.get("Retry-After")
 569                          err = RetryableError("Rate limit exceeded - please try again later")
 570                          if retry_after_header:
 571                              try:
 572                                  err.retry_after = float(retry_after_header)
 573                              except (ValueError, TypeError):
 574                                  pass
 575                          raise err
 576                      elif response.status >= 500:
 577                          # A-P2-1: 5xx 为可重试错误
 578                          raise RetryableError(f"Server error: {response.status}")
 579                      else:
 580                          error_text = await response.text()
 581                          # P1-006修复: 处理token超限错误
 582                          if "token" in error_text.lower() or "length" in error_text.lower():
 583                              raise NonRetryableError(f"Token limit exceeded: {error_text}")
 584                          # A-P2-1: 其他 4xx 为不可重试错误
 585                          raise NonRetryableError(f"GLM API error: {response.status} - {error_text}")
 586          except asyncio.TimeoutError:
 587              # A-P2-1: 超时为可重试错误
 588              raise RetryableError("API request timed out after 180s")
 589          except aiohttp.ClientError as e:
 590              # A-P2-1: 网络错误为可重试错误
 591              raise RetryableError(f"Network error: {e}")

 594  class ConversationMemory:
 595      """
 596      对话历史管理器
 597      
 598      功能：
 599      - 基于会话 ID 的对话历史存储
 600      - 支持 Redis 持久化（可选）
 601      - 自动过期清理
 602      - 上下文窗口管理
 603      """
 604      
 605      def __init__(
 606          self,
 607          max_history_turns: int = 10,
 608          ttl_seconds: int = 3600,
 609          use_redis: bool = False,
 610          redis_client = None
 611      ):
 612          self.max_history_turns = max_history_turns
 613          self.ttl_seconds = ttl_seconds
 614          self.use_redis = use_redis
 615          self.redis_client = redis_client
 616          self._local_cache: Dict[str, List[Dict[str, str]]] = {}
 617          self._lock = threading.RLock()
 618      
 619      def _get_redis_key(self, session_id: str) -> str:
 620          return f"stylist:conversation:{session_id}"
 621      
 622      async def get_history(self, session_id: str) -> List[Dict[str, str]]:
 623          """
 624          获取会话的对话历史
 625          
 626          Args:
 627              session_id: 会话 ID
 628              
 629          Returns:
 630              对话历史列表
 631          """
 632          if self.use_redis and self.redis_client:
 633              try:
 634                  key = self._get_redis_key(session_id)
 635                  data = await self.redis_client.get(key)
 636                  if data:
 637                      return json.loads(data)
 638              except Exception as e:
 639                  logger.warning(f"Redis get failed for session {session_id}: {e}")
 640          
 641          with self._lock:
 642              return self._local_cache.get(session_id, [])
 643      
 644      async def add_message(
 645          self,
 646          session_id: str,
 647          role: str,
 648          content: str
 649      ) -> List[Dict[str, str]]:
 650          """
 651          添加消息到对话历史
 652          
 653          Args:
 654              session_id: 会话 ID
 655              role: 角色 (user/assistant/system)
 656              content: 消息内容
 657              
 658          Returns:
 659              更新后的对话历史
 660          """
 661          with self._lock:
 662              history = self._local_cache.get(session_id, [])
 663              
 664              # 添加新消息
 665              history.append({
 666                  "role": role,
 667                  "content": content,
 668                  "timestamp": datetime.now().isoformat()
 669              })
 670              
 671              # 保持历史记录在限制内（保留最近 N 轮对话）
 672              max_messages = self.max_history_turns * 2
 673              if len(history) > max_messages:
 674                  history = history[-max_messages:]
 675              
 676              self._local_cache[session_id] = history
 677          
 678          # 持久化到 Redis
 679          if self.use_redis and self.redis_client:
 680              try:
 681                  key = self._get_redis_key(session_id)
 682                  await self.redis_client.setex(
 683                      key,
 684                      self.ttl_seconds,
 685                      json.dumps(history)
 686                  )
 687              except Exception as e:
 688                  logger.warning(f"Redis set failed for session {session_id}: {e}")
 689          
 690          return history
 691      
 692      async def clear_history(self, session_id: str) -> None:
 693          """
 694          清除会话的对话历史
 695          
 696          Args:
 697              session_id: 会话 ID
 698          """
 699          with self._lock:
 700              if session_id in self._local_cache:
 701                  del self._local_cache[session_id]
 702          
 703          if self.use_redis and self.redis_client:
 704              try:
 705                  key = self._get_redis_key(session_id)
 706                  await self.redis_client.delete(key)
 707              except Exception as e:
 708                  logger.warning(f"Redis delete failed for session {session_id}: {e}")
 709      
 710      def get_stats(self) -> Dict[str, Any]:
 711          """获取统计信息"""
 712          with self._lock:
 713              return {
 714                  "active_sessions": len(self._local_cache),
 715                  "total_messages": sum(len(h) for h in self._local_cache.values()),
 716                  "max_history_turns": self.max_history_turns,
 717                  "ttl_seconds": self.ttl_seconds,
 718                  "use_redis": self.use_redis
 719              }

1138      def _compute_cache_key(self, prefix: str, *args, **kwargs) -> str:
1139          """Compute cache key for method calls"""
1140          key_data = json.dumps({"args": args, "kwargs": kwargs}, sort_keys=True, default=str)
1141          key_hash = hashlib.sha256(key_data.encode('utf-8')).hexdigest()
1142          return f"{prefix}:{key_hash}"
