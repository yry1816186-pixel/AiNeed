 134          self,
 135          candidates: List[ClothingItem],
 136          user_profile: UserProfile,
 137          context: OutfitContext,
 138      ) -> Optional[ClothingItem]:
 139          """选择核心单品（anchor piece）
 140  
 141          核心单品是整套搭配的锚点，通常是：
 142          - 正式场合：外套/上装
 143          - 日常休闲：上装或连衣裙
 144          - 冬季：外套
 145  
 146          Args:
 147              candidates: 候选单品列表
 148              user_profile: 用户档案
 149              context: 搭配上下文
 150  
 151          Returns:
 152              最适合的核心单品，或 None
 153          """
 154          scored = self._pick_anchor_candidates(
 155              candidates, user_profile, context, [], 0.5
 156          )
 157          return scored[0] if scored else None

 159      def _pick_anchor_candidates(
 160          self,
 161          candidates: List[ClothingItem],
 162          user_profile: UserProfile,
 163          context: OutfitContext,
 164          target_styles: List[str],
 165          target_formality: float,
 166      ) -> List[ClothingItem]:
 167          """选择多个候选核心单品
 168  
 169          策略：
 170          1. 根据场合确定锚点品类优先级
 171          2. 对每个候选打分：体型匹配 + 场合匹配 + 季节匹配 + 风格匹配
 172          3. 返回按分排序的候选列表
 173          """
 174          # 确定锚点品类优先级
 175          anchor_categories = self._get_anchor_categories(context)
 176  
 177          # 筛选合适品类的候选
 178          filtered = [
 179              c for c in candidates if c.category in anchor_categories
 180          ]
 181  
 182          if not filtered:
 183              # 回退：任意品类
 184              filtered = list(candidates)
 185  
 186          scored_list: List[Tuple[float, ClothingItem]] = []
 187          for item in filtered:
 188              score = 0.0
 189  
 190              # 体型匹配（权重 0.25）
 191              body_score = self._item_body_fit_score(item, user_profile)
 192              score += body_score * 0.25
 193  
 194              # 场合匹配（权重 0.25）
 195              occasion_score = self._item_occasion_score(item, context)
 196              score += occasion_score * 0.25
 197  
 198              # 正式度匹配（权重 0.15）
 199              formality_diff = abs(item.formality - target_formality)
 200              formality_score = max(0.0, 1.0 - formality_diff)
 201              score += formality_score * 0.15
 202  
 203              # 风格匹配（权重 0.20）
 204              style_score = self._item_style_score(item, target_styles)
 205              score += style_score * 0.20
 206  
 207              # 季节匹配（权重 0.10）
 208              season_score = self._item_season_score(item, context)
 209              score += season_score * 0.10
 210  
 211              # 价格合理性（权重 0.05）：不是最贵也不是最便宜
 212              if item.price > 0:
 213                  price_score = 0.5  # 中等价格加分
 214                  score += price_score * 0.05
 215  
 216              scored_list.append((score, item))
 217  
 218          scored_list.sort(key=lambda x: x[0], reverse=True)
 219          return [item for _, item in scored_list]

 221      def _get_anchor_categories(self, context: OutfitContext) -> List[str]:
 222          """根据场合确定锚点品类优先级"""
 223          occasion = context.occasion
 224          temp = context.temperature_celsius
 225          season = context.season or _get_season_from_month(datetime.now().month)
 226  
 227          # 冬季/低温：外套优先
 228          if (temp is not None and temp < 10) or season == "winter":
 229              if occasion in ("interview", "business", "work", "wedding", "ceremony"):
 230                  return ["outerwear", "tops", "dresses"]
 231              return ["outerwear", "tops", "dresses"]
 232  
 233          # 正式场合：上装/连衣裙优先
 234          if occasion in ("interview", "business", "wedding", "ceremony"):
 235              return ["tops", "outerwear", "dresses"]
 236  
 237          # 约会/社交：连衣裙或上装
 238          if occasion in ("date", "party"):
 239              return ["dresses", "tops", "outerwear"]
 240  
 241          # 日常/旅行
 242          return ["tops", "dresses", "outerwear"]

 248      def _expand_outfit_chain(
 249          self,
 250          anchor: ClothingItem,
 251          slots: List[ClothingSlot],
 252          user_profile: UserProfile,
 253          context: OutfitContext,
 254          candidates: List[ClothingItem],
 255          budget: float,
 256          target_styles: List[str],
 257          target_colors: List[str],
 258          target_formality: float,
 259      ) -> Optional[FullOutfitPlan]:
 260          """基于核心单品扩展完整搭配
 261  
 262          逐槽位填充，每个槽位选择与已选单品兼容度最高的候选。
 263          """
 264          plan = FullOutfitPlan(
 265              plan_id=f"plan_{uuid.uuid4().hex[:12]}",
 266              anchor_piece=anchor,
 267          )
 268  
 269          selected_items: List[ClothingItem] = [anchor]
 270          remaining_budget = budget
 271          used_ids: Set[str] = {anchor.item_id}
 272  
 273          # 确定锚点是什么品类，调整需要填充的槽位
 274          actual_slots = self._adjust_slots_for_anchor(slots, anchor)
 275  
 276          for slot in actual_slots:
 277              slot_category = self._slot_to_category(slot)
 278              if not slot_category:
 279                  continue
 280  
 281              # 从候选中筛选品类匹配
 282              slot_candidates = [
 283                  c for c in candidates
 284                  if c.category in slot_category
 285                  and c.item_id not in used_ids
 286              ]
 287  
 288              if not slot_candidates:
 289                  continue
 290  
 291              # 为每个候选计算与已选单品的兼容度
 292              best_item: Optional[ClothingItem] = None
 293              best_score = -1.0
 294  
 295              for candidate in slot_candidates:
 296                  # 预算检查
 297                  if remaining_budget > 0 and candidate.price > remaining_budget:
 298                      continue
 299  
 300                  compat = self._item_compatibility_score(
 301                      candidate, selected_items, user_profile, context,
 302                      target_styles, target_colors, target_formality,
 303                  )
 304                  if compat > best_score:
 305                      best_score = compat
 306                      best_item = candidate
 307  
 308              if best_item is not None:
 309                  self._assign_slot_to_plan(plan, slot, best_item)
 310                  selected_items.append(best_item)
 311                  remaining_budget -= best_item.price
 312                  used_ids.add(best_item.item_id)
 313  
 314          return plan

 316      def _determine_slots(
 317          self, context: OutfitContext, user_profile: UserProfile
 318      ) -> List[ClothingSlot]:
 319          """确定搭配槽位列表
 320  
 321          根据场合和温度决定需要的层次和配饰。
 322          """
 323          slots: List[ClothingSlot] = []
 324          temp = context.temperature_celsius
 325          season = context.season or _get_season_from_month(datetime.now().month)
 326  
 327          # 低温层搭
 328          if temp is not None and temp < 10:
 329              slots.extend([ClothingSlot.OUTER_TOP, ClothingSlot.INNER_TOP])
 330          elif temp is not None and temp < 20:
 331              slots.append(ClothingSlot.OUTER_TOP)
 332          else:
 333              slots.append(ClothingSlot.INNER_TOP)
 334  
 335          # 下装 + 鞋子（始终需要）
 336          slots.extend([ClothingSlot.BOTTOM, ClothingSlot.SHOES])
 337  
 338          # 配饰（根据场合添加）
 339          occasion = context.occasion
 340          if occasion in ("date", "party", "wedding"):
 341              slots.extend([
 342                  ClothingSlot.BAG,
 343                  ClothingSlot.NECKLACE,
 344                  ClothingSlot.EARRINGS,
 345              ])
 346          elif occasion in ("interview", "business", "work"):
 347              slots.extend([ClothingSlot.BAG, ClothingSlot.WATCH])
 348          elif occasion in ("travel", "daily", "campus"):
 349              slots.extend([ClothingSlot.BAG, ClothingSlot.SUNGLASSES])
 350          else:
 351              slots.extend([ClothingSlot.BAG])
 352  
 353          # 冬季额外配饰
 354          if season == "winter" or (temp is not None and temp < 5):
 355              slots.append(ClothingSlot.SCARF)
 356  
 357          return slots

 359      def _adjust_slots_for_anchor(
 360          self, slots: List[ClothingSlot], anchor: ClothingItem
 361      ) -> List[ClothingSlot]:
 362          """根据锚点品类调整槽位
 363  
 364          如果锚点是连衣裙，则不需要下装和内搭上装。
 365          如果锚点是外套，则不再需要外层。
 366          """
 367          adjusted: List[ClothingSlot] = []
 368          anchor_cat = anchor.category
 369  
 370          # 连衣裙特殊处理
 371          if anchor_cat == "dresses":
 372              for slot in slots:
 373                  if slot in (ClothingSlot.INNER_TOP, ClothingSlot.OUTER_TOP, ClothingSlot.BOTTOM):
 374                      continue
 375                  adjusted.append(slot)
 376              # 连衣裙场合需要外套
 377              adjusted.insert(0, ClothingSlot.OUTER_TOP)
 378              return adjusted
 379  
 380          # 外套作为锚点：不需要另一个外套，需要内搭
 381          if anchor_cat == "outerwear":
 382              for slot in slots:
 383                  if slot == ClothingSlot.OUTER_TOP:
 384                      continue
 385                  adjusted.append(slot)
 386              if ClothingSlot.INNER_TOP not in adjusted:
 387                  adjusted.insert(0, ClothingSlot.INNER_TOP)
 388              return adjusted
 389  
 390          # 上装作为锚点：不需要内搭上装
 391          if anchor_cat == "tops":
 392              for slot in slots:
 393                  if slot == ClothingSlot.INNER_TOP:
 394                      continue
 395                  adjusted.append(slot)
 396              return adjusted
 397  
 398          return list(slots)

 416              ClothingSlot.SUNGLASSES: ["accessories"],

 436      # Compatibility scoring (single item vs. already-selected items)
 437      # ------------------------------------------------------------------

 439      def _item_compatibility_score(
 440          self,
 441          candidate: ClothingItem,
 442          selected_items: List[ClothingItem],
 443          user_profile: UserProfile,
 444          context: OutfitContext,
 445          target_styles: List[str],
 446          target_colors: List[str],
 447          target_formality: float,
 448      ) -> float:
 449          """计算候选单品与已选单品的综合兼容度
 450  
 451          维度权重：
 452          - 色彩兼容 0.25
 453          - 风格兼容 0.25
 454          - 场合/正式度 0.15
 455          - 体型适配 0.15
 456          - 季节适配 0.10
 457          - 价格合理 0.10
 458          """
 459          score = 0.0
 460  
 461          # 1. 色彩兼容
 462          color_compat = self._color_compatibility_with_existing(
 463              candidate, selected_items, target_colors
 464          )
 465          score += color_compat * 0.25
 466  
 467          # 2. 风格兼容
 468          style_compat = self._style_compatibility_with_existing(
 469              candidate, selected_items, target_styles
 470          )
 471          score += style_compat * 0.25
 472  
 473          # 3. 场合/正式度
 474          occasion_score = self._item_occasion_score(candidate, context)
 475          formality_diff = abs(candidate.formality - target_formality)
 476          formality_score = max(0.0, 1.0 - formality_diff * 1.5)
 477          score += (occasion_score * 0.5 + formality_score * 0.5) * 0.15
 478  
 479          # 4. 体型适配
 480          body_score = self._item_body_fit_score(candidate, user_profile)
 481          score += body_score * 0.15
 482  
 483          # 5. 季节适配
 484          season_score = self._item_season_score(candidate, context)
 485          score += season_score * 0.10
 486  
 487          # 6. 价格合理（价格适中的加分）
 488          if candidate.price > 0:
 489              price_score = 0.5
 490              if len(selected_items) > 0:
 491                  avg_price = sum(i.price for i in selected_items) / len(selected_items)
 492                  # 与已选单品价格水平接近的加分
 493                  price_ratio = min(candidate.price, avg_price) / max(candidate.price, avg_price, 1)
 494                  price_score = 0.3 + price_ratio * 0.7
 495              score += price_score * 0.10
 496  
 497          return score

 503      def _check_color_harmony(self, items: List[ClothingItem]) -> ColorHarmonyResult:
 504          """检查整体色彩协调性
 505  
 506          规则：
 507          - 不超过 3 个主色调（中性色不算）
 508          - 主色(60%) + 辅助色(30%) + 点缀色(10%)
 509          - 使用色环理论判断色彩搭配类型
 510          - 每种色彩方案有对应的场合适用性
 511  
 512          Args:
 513              items: 所有单品列表
 514  
 515          Returns:
 516              色彩协调检查结果
 517          """
 518          if not items:
 519              return ColorHarmonyResult(
 520                  is_harmonious=False,
 521                  score=0.0,
 522                  issues=["没有单品"],
 523              )
 524  
 525          # 收集所有颜色
 526          color_families: Dict[str, int] = {}
 527          for item in items:
 528              primary = _resolve_color_family(item.color_primary)
 529              color_families[primary] = color_families.get(primary, 0) + 1
 530              if item.color_secondary:
 531                  secondary = _resolve_color_family(item.color_secondary)
 532                  color_families[secondary] = color_families.get(secondary, 0) + 1
 533  
 534          # 去掉中性色，统计非中性色族
 535          non_neutral = {k: v for k, v in color_families.items() if k != "neutral"}
 536          neutral_count = color_families.get("neutral", 0)
 537          total_colors = sum(color_families.values())
 538  
 539          # 1. 主色调数量检查
 540          issues: List[str] = []
 541          if len(non_neutral) > 3:
 542              issues.append(
 543                  f"色彩过多（{len(non_neutral)}个主色调），建议控制在3个以内"
 544              )
 545  
 546          # 2. 色彩比例检查（60-30-10 法则）
 547          primary_ratio_score = 0.0
 548          if non_neutral:
 549              sorted_families = sorted(non_neutral.items(), key=lambda x: x[1], reverse=True)
 550              dominant_count = sorted_families[0][1]
 551              dominant_ratio = dominant_count / max(total_colors, 1)
 552  
 553              # 主色应占约 50-70%
 554              if 0.4 <= dominant_ratio <= 0.7:
 555                  primary_ratio_score = 1.0
 556              elif 0.3 <= dominant_ratio <= 0.8:
 557                  primary_ratio_score = 0.7
 558              else:
 559                  primary_ratio_score = 0.4
 560                  issues.append("主色比例不够突出，建议调整为主要颜色占60%")
 561  
 562          # 3. 色环协调检查
 563          harmony_type = self._classify_color_scheme(non_neutral)
 564          harmony_score = self._score_color_scheme_type(harmony_type, non_neutral)
 565  
 566          # 4. 中性色搭配检查
 567          neutral_score = 0.5
 568          if neutral_count > 0 and len(non_neutral) > 0:
 569              # 有中性色做平衡是好的
 570              neutral_score = 0.7 + min(neutral_count / total_colors, 0.3)
 571          elif neutral_count > 0 and len(non_neutral) == 0:
 572              # 纯中性色搭配
 573              neutral_score = 0.8
 574              harmony_type = ColorSchemeType.ACHROMATIC
 575  
 576          # 综合评分
 577          final_score = (
 578              (primary_ratio_score * 0.35)
 579              + (harmony_score * 0.45)
 580              + (neutral_score * 0.20)
 581          )
 582  
 583          # 如果有超过3个非中性色，扣分
 584          if len(non_neutral) > 3:
 585              final_score *= max(0.5, 1.0 - (len(non_neutral) - 3) * 0.15)
 586  
 587          is_harmonious = final_score >= 0.6 and len(issues) == 0
 588  
 589          dominant_colors = list(non_neutral.keys())[:3]
 590          accent_colors = list(non_neutral.keys())[3:] if len(non_neutral) > 3 else []
 591  
 592          suggestion = self._color_harmony_suggestion(harmony_type, issues)
 593  
 594          return ColorHarmonyResult(
 595              is_harmonious=is_harmonious,
 596              scheme_type=harmony_type,
 597              dominant_colors=dominant_colors,
 598              accent_colors=accent_colors,
 599              score=round(final_score * 100, 2),
 600              issues=issues,
 601              suggestion=suggestion,

 604      def _classify_color_scheme(self, non_neutral: Dict[str, int]) -> ColorSchemeType:
 605          """根据色环分类色彩方案类型"""
 606          families = list(non_neutral.keys())
 607          if len(families) == 0:
 608              return ColorSchemeType.ACHROMATIC
 609          if len(families) == 1:
 610              return ColorSchemeType.MONOCHROMATIC
 611  
 612          # 计算所有色族之间的色环距离
 613          hues = [_FAMILY_HUE.get(f, 0.0) for f in families if f != "neutral"]
 614  
 615          if len(hues) < 2:
 616              return ColorSchemeType.NEUTRAL
 617  
 618          # 检查各种配色类型
 619          distances = []
 620          for i in range(len(hues)):
 621              for j in range(i + 1, len(hues)):
 622                  dist = abs(hues[i] - hues[j])
 623                  dist = min(dist, 360.0 - dist)
 624                  distances.append(dist)
 625  
 626          avg_dist = sum(distances) / len(distances) if distances else 0
 627  
 628          # 互补色（约180度）
 629          if any(160 < d < 200 for d in distances):
 630              if len(families) == 2:
 631                  return ColorSchemeType.COMPLEMENTARY
 632              # 检查分裂互补
 633              if len(families) == 3:
 634                  small_dists = sorted(distances)[:2]
 635                  if all(d < 60 for d in small_dists):
 636                      return ColorSchemeType.SPLIT_COMPLEMENTARY
 637              return ColorSchemeType.COMPLEMENTARY
 638  
 639          # 三角配色（约120度间隔）
 640          if len(families) == 3 and all(100 < d < 140 for d in distances):
 641              return ColorSchemeType.TRIADIC
 642  
 643          # 类似色（相邻色环，<60度）
 644          if all(d < 60 for d in distances):
 645              return ColorSchemeType.ANALOGOUS
 646  
 647          # 默认归为类似色
 648          return ColorSchemeType.ANALOGOUS

 686      def _color_compatibility_with_existing(
 687          self,
 688          candidate: ClothingItem,
 689          selected_items: List[ClothingItem],
 690          target_colors: List[str],
 691      ) -> float:
 692          """计算候选单品与已选单品的色彩兼容度"""
 693          cand_family = _resolve_color_family(candidate.color_primary)
 694          if cand_family == "neutral":
 695              return 0.9  # 中性色百搭
 696  
 697          score = 0.0
 698          for existing in selected_items:
 699              ex_family = _resolve_color_family(existing.color_primary)
 700              if ex_family == "neutral":
 701                  score += 0.85
 702                  continue
 703              dist = _hue_distance(cand_family, ex_family)
 704              # 距离越小越和谐，同类色最高分
 705              if dist < 30:
 706                  score += 0.9
 707              elif dist < 60:
 708                  score += 0.8
 709              elif dist < 120:
 710                  score += 0.6
 711              elif dist < 160:
 712                  score += 0.4
 713              else:
 714                  # 互补色需要技巧，给中等分
 715                  score += 0.5
 716  
 717          if selected_items:
 718              score /= len(selected_items)
 719  
 720          # 如果候选颜色在用户目标颜色中，加分
 721          if target_colors:
 722              target_families = [_resolve_color_family(c) for c in target_colors]
 723              if cand_family in target_families:
 724                  score = min(1.0, score + 0.1)
 725  
 726          return min(1.0, score)

 765      def _check_style_consistency(self, items: List[ClothingItem]) -> StyleConsistencyResult:
 766          """检查风格一致性
 767  
 768          规则：
 769          - 相邻风格可混搭
 770          - 对立风格不可（如 formal + sporty 不行）
 771          - 使用兼容矩阵量化风格冲突
 772  
 773          Args:
 774              items: 所有单品列表
 775  
 776          Returns:
 777              风格一致性检查结果
 778          """
 779          if not items:
 780              return StyleConsistencyResult(score=0.0)
 781  
 782          # 收集所有风格标签
 783          all_styles: Dict[str, int] = {}
 784          for item in items:
 785              for tag in item.style_tags:
 786                  tag_lower = tag.lower().replace(" ", "_").replace("-", "_")
 787                  all_styles[tag_lower] = all_styles.get(tag_lower, 0) + 1
 788  
 789          if not all_styles:
 790              # 无风格标签，默认中等一致性
 791              return StyleConsistencyResult(
 792                  is_consistent=True,
 793                  dominant_style="unknown",
 794                  score=60.0,
 795              )
 796  
 797          # 找主导风格
 798          sorted_styles = sorted(all_styles.items(), key=lambda x: x[1], reverse=True)
 799          dominant_style = sorted_styles[0][0]
 800  
 801          # 检查风格冲突
 802          conflicts: List[Tuple[str, str]] = []
 803          issues: List[str] = []
 804          total_compat = 0.0
 805          pair_count = 0
 806  
 807          style_list = list(all_styles.keys())
 808          for i in range(len(style_list)):
 809              for j in range(i + 1, len(style_list)):
 810                  s1 = style_list[i]
 811                  s2 = style_list[j]
 812                  compat = self._get_style_compatibility(s1, s2)
 813                  total_compat += compat
 814                  pair_count += 1
 815  
 816                  if compat < 0.2:
 817                      conflicts.append((s1, s2))
 818                      issues.append(f"风格冲突：{s1} 与 {s2} 不搭")
 819                  elif compat < 0.4:
 820                      issues.append(f"风格轻微不协调：{s1} 与 {s2}")
 821  
 822          avg_compat = total_compat / pair_count if pair_count > 0 else 0.5
 823  
 824          # 评分：兼容度转化为 0-100
 825          score = avg_compat * 100.0
 826  
 827          # 有严重冲突扣分
 828          if conflicts:
 829              score = max(0, score - len(conflicts) * 15)
 830  
 831          is_consistent = len(conflicts) == 0 and score >= 60
 832  
 833          return StyleConsistencyResult(
 834              is_consistent=is_consistent,
 835              dominant_style=dominant_style,
 836              style_conflict=conflicts,
 837              score=round(score, 2),
 838              issues=issues,
 839          )

 841      def _get_style_compatibility(self, style_a: str, style_b: str) -> float:
 842          """从兼容矩阵查询两个风格的兼容度"""
 843          if style_a == style_b:
 844              return 1.0
 845          compat = _STYLE_COMPATIBILITY.get(style_a, {}).get(style_b, 0.3)
 846          return compat

 848      def _style_compatibility_with_existing(
 849          self,
 850          candidate: ClothingItem,
 851          selected_items: List[ClothingItem],
 852          target_styles: List[str],
 853      ) -> float:
 854          """计算候选单品与已选单品的风格兼容度"""
 855          cand_styles = [s.lower().replace(" ", "_").replace("-", "_") for s in candidate.style_tags]
 856          if not cand_styles:
 857              return 0.5
 858  
 859          total_score = 0.0
 860          comparisons = 0
 861  
 862          for existing in selected_items:
 863              ex_styles = [s.lower().replace(" ", "_").replace("-", "_") for s in existing.style_tags]
 864              if not ex_styles:
 865                  continue
 866  
 867              # 计算候选风格与已有风格的最大兼容度
 868              best_compat = 0.0
 869              for cs in cand_styles:
 870                  for es in ex_styles:
 871                      compat = self._get_style_compatibility(cs, es)
 872                      best_compat = max(best_compat, compat)
 873  
 874              total_score += best_compat
 875              comparisons += 1
 876  
 877          avg = total_score / comparisons if comparisons > 0 else 0.5
 878  
 879          # 如果候选风格在目标风格中，加分
 880          if target_styles:
 881              target_lower = [s.lower() for s in target_styles]
 882              matching = [s for s in cand_styles if s in target_lower]
 883              if matching:
 884                  avg = min(1.0, avg + 0.15)
 885  
 886          return min(1.0, avg)

 892      def _check_weather_suitability(
 893          self, items: List[ClothingItem], temperature_celsius: float
 894      ) -> WeatherResult:
 895          """检查天气适配性
 896  
 897          规则：
 898          - <10C: 需要3层（内搭+中间层+外套），优先保暖材质
 899          - 10-20C: 2层（内搭+外套），可调节
 900          - 20-25C: 1-2层，轻薄材质
 901          - >25C: 1层，透气材质
 902  
 903          Args:
 904              items: 所有单品列表
 905              temperature_celsius: 当前温度
 906  
 907          Returns:
 908              天气适配检查结果
 909          """
 910          if not items:
 911              return WeatherResult(is_suitable=False, score=0.0, issues=["没有单品"])
 912  
 913          issues: List[str] = []
 914          total_warmth = sum(item.warmth_level for item in items)
 915          layer_count = len([i for i in items if i.category in ("tops", "outerwear", "dresses")])
 916  
 917          # 确定期望的层次和保暖度
 918          if temperature_celsius < 0:
 919              expected_layers = 3
 920              expected_warmth_min = 7
 921              expected_warmth_max = 12
 922              temp_label = "极寒"
 923          elif temperature_celsius < 10:
 924              expected_layers = 3
 925              expected_warmth_min = 5
 926              expected_warmth_max = 10
 927              temp_label = "寒冷"
 928          elif temperature_celsius < 15:
 929              expected_layers = 2
 930              expected_warmth_min = 3
 931              expected_warmth_max = 7
 932              temp_label = "较冷"
 933          elif temperature_celsius < 20:
 934              expected_layers = 2
 935              expected_warmth_min = 2
 936              expected_warmth_max = 5
 937              temp_label = "凉爽"
 938          elif temperature_celsius < 25:
 939              expected_layers = 1
 940              expected_warmth_min = 1
 941              expected_warmth_max = 3
 942              temp_label = "温暖"
 943          elif temperature_celsius < 30:
 944              expected_layers = 1
 945              expected_warmth_min = 0
 946              expected_warmth_max = 2
 947              temp_label = "炎热"
 948          else:
 949              expected_layers = 1
 950              expected_warmth_min = 0
 951              expected_warmth_max = 1
 952              temp_label = "酷热"
 953  
 954          # 层次检查
 955          layer_score = 0.0
 956          if layer_count >= expected_layers:
 957              layer_score = 1.0
 958          elif layer_count == expected_layers - 1:
 959              layer_score = 0.7
 960              if expected_layers > 1:
 961                  issues.append(f"当前{temp_label}（{temperature_celsius:.0f}C），建议增加一层穿搭")
 962          else:
 963              layer_score = 0.3
 964              issues.append(f"当前{temp_label}（{temperature_celsius:.0f}C），穿衣层数不足")
 965  
 966          # 保暖度检查
 967          warmth_score = 0.0
 968          if expected_warmth_min <= total_warmth <= expected_warmth_max:
 969              warmth_score = 1.0
 970          elif total_warmth < expected_warmth_min:
 971              warmth_score = max(0.2, total_warmth / expected_warmth_min)
 972              issues.append(f"保暖度不足（当前{total_warmth}，建议{expected_warmth_min}+）")
 973          elif total_warmth > expected_warmth_max:
 974              excess_ratio = expected_warmth_max / total_warmth
 975              warmth_score = max(0.3, excess_ratio)
 976              issues.append(f"可能过热（保暖度{total_warmth}，建议{expected_warmth_max}以内）")
 977  
 978          final_score = layer_score * 0.5 + warmth_score * 0.5
 979          final_score *= 100
 980  
 981          is_suitable = len(issues) == 0 and final_score >= 60
 982  
 983          suggestion = ""
 984          if issues:
 985              suggestion = f"当前{temp_label}（{temperature_celsius:.0f}C），" + "；".join(issues)
 986  
 987          return WeatherResult(
 988              is_suitable=is_suitable,
 989              layer_count=layer_count,
 990              warmth_total=total_warmth,
 991              score=round(final_score, 2),
 992              issues=issues,
 993              suggestion=suggestion,
 994          )

1048              elif ratio <= 1.1:
1049                  score = 50.0
1050              else:

1054          if not is_within:
1055              issues.append(f"超出预算：总价{total_price:.0f}元，预算{budget:.0f}元")

1058              is_within_budget=is_within,
1059              total_price=total_price,

1164          - 风格一致 25%

1175  
1176          # 风格一致
1177          style = self._check_style_consistency(items)
1178  
1179          # 体型适配
1180          body = self._check_body_fit(items, user_profile)
1181  
1182          # 天气
1183          if context.temperature_celsius is not None:
1184              weather = self._check_weather_suitability(items, context.temperature_celsius)
1185              weather_score = weather.score
1186          else:
1187              weather_score = self._estimate_season_score(items, context.season or _get_season_from_month(datetime.now().month))
1188  
1189          # 场合
1190          occasion_score = self._overall_occasion_score(items, context) * 100
1191  
1192          # 价格合理性（单品的场合匹配度而非预算检查）
1193          price_score = 70.0  # 默认中等
1194  
1195          overall = (
1196              color.score * 0.25
1197              + style.score * 0.25
1198              + body.score * 0.20
1199              + weather_score * 0.15
1200              + occasion_score * 0.10
1201              + price_score * 0.05
1202          )
1203  
1204          return round(min(100.0, overall), 2)
1205  
1206      # ------------------------------------------------------------------
1207      # Occasion scoring helpers
1208      # ------------------------------------------------------------------
1209  
1210      def _item_occasion_score(
1211          self, item: ClothingItem, context: OutfitContext
1212      ) -> float:
1213          """计算单品与场合的匹配度"""
1214          occasion = context.occasion
1215          if not occasion:
1216              return 0.6
1217  
1218          # 检查 item.occasions 字段
1219          if item.occasions and occasion in item.occasions:
1220              return 1.0
1221          # 模糊匹配
1222          if item.occasions:
1223              for occ in item.occasions:
1224                  if occ in occasion or occasion in occ:
1225                      return 0.8
1226  
1227          # 基于正式度推断
1228          target_formality = _OCCASION_FORMALITY.get(occasion, 0.5)
1229          formality_diff = abs(item.formality - target_formality)
1230          return max(0.2, 1.0 - formality_diff * 1.5)
1231  
1232      def _overall_occasion_score(
1233          self, items: List[ClothingItem], context: OutfitContext
1234      ) -> float:
1235          """计算整套搭配与场合的整体匹配度"""
1236          if not items:
1237              return 0.0
1238          total = sum(self._item_occasion_score(item, context) for item in items)
1239          return total / len(items)

1241      def _item_season_score(
1242          self, item: ClothingItem, context: OutfitContext
1243      ) -> float:
1244          """计算单品与季节的匹配度"""
1245          season = context.season or _get_season_from_month(datetime.now().month)
1246  
1247          if item.seasons and season in item.seasons:
1248              return 1.0
1249          if item.seasons:
1250              return 0.3
1251          return 0.5

1253      def _item_style_score(
1254          self, item: ClothingItem, target_styles: List[str]
1255      ) -> float:
1256          """计算单品风格与目标风格的匹配度"""
1257          if not target_styles:
1258              return 0.5
1259  
1260          item_lower = [s.lower().replace(" ", "_").replace("-", "_") for s in item.style_tags]
1261          target_lower = [s.lower().replace(" ", "_").replace("-", "_") for s in target_styles]
1262  
1263          if not item_lower:
1264              return 0.4
1265  
1266          matching = len(set(item_lower) & set(target_lower))
1267          return min(1.0, matching / max(len(target_lower), 1) + 0.3)

1328      def _resolve_formality(
1329          self, user_profile: UserProfile, context: OutfitContext
1330      ) -> float:
1331          """确定目标正式度"""
1332          if context.formality_hint is not None:
1333              return context.formality_hint
1334  
1335          occasion_formality = _OCCASION_FORMALITY.get(context.occasion, 0.5)
1336  
1337          # 如果用户有风格偏好，取中间值
1338          if user_profile.style_preferences:
1339              user_formalities = [
1340                  _STYLE_FORMALITY.get(s.lower(), 0.5)
1341                  for s in user_profile.style_preferences
1342              ]
1343              user_avg = sum(user_formalities) / len(user_formalities)
1344              return (occasion_formality + user_avg) / 2
1345  
1346          return occasion_formality

1425      def _extract_dominant_styles(self, items: List[ClothingItem]) -> List[str]:
1426          """提取方案的主导风格标签"""
1427          style_counts: Dict[str, int] = {}
1428          for item in items:

1430                  style_counts[tag] = style_counts.get(tag, 0) + 1
1431          sorted_styles = sorted(style_counts.items(), key=lambda x: x[1], reverse=True)
