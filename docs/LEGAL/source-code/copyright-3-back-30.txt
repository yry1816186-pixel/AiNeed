1439      def get_outfit_combination_score(self, outfit_items: List[Dict[str, Any]],
1440                                       body_profile: BodyProfile) -> Dict[str, Any]:
1441          """
1442          计算整套穿搭的适配分数
1443  
1444          Args:
1445              outfit_items: 穿搭单品列表
1446              body_profile: 用户身体档案
1447  
1448          Returns:
1449              Dict: 整套穿搭的适配分析结果
1450          """
1451          if not outfit_items:
1452              return {
1453                  "overall_score": 0.0,
1454                  "items_analysis": [],
1455                  "recommendation": "无穿搭数据",
1456              }
1457  
1458          items_analysis = []
1459          total_score = 0.0
1460          category_scores = {}
1461  
1462          for item in outfit_items:
1463              fit_score = self.calculate_fit_score(item, body_profile)
1464              category = item.get("category", "unknown")
1465  
1466              items_analysis.append({
1467                  "item_id": item.get("id", ""),
1468                  "item_name": item.get("name", ""),
1469                  "category": category,
1470                  "fit_score": fit_score.overall_score,
1471                  "scores": {
1472                      "cut_match": fit_score.cut_match_score,
1473                      "color_match": fit_score.color_match_score,
1474                      "style_match": fit_score.style_match_score,
1475                      "body_optimization": fit_score.body_type_score,
1476                  },
1477              })
1478  
1479              category_scores[category] = fit_score.overall_score
1480              total_score += fit_score.overall_score
1481  
1482          # 计算平均分数
1483          avg_score = total_score / len(outfit_items)
1484  
1485          # 计算整体协调性（各单品分数的方差越小越好）
1486          scores_list = [item["fit_score"] for item in items_analysis]
1487          variance = sum((s - avg_score) ** 2 for s in scores_list) / len(scores_list)
1488          harmony_bonus = max(0, 0.1 - variance) * 5  # 协调性加分
1489  
1490          final_score = min(1.0, avg_score + harmony_bonus)
1491  
1492          return {

1494              "average_item_score": round(avg_score, 3),
1495              "harmony_bonus": round(harmony_bonus, 3),
1496              "items_analysis": items_analysis,

1402              for key, cached in reversed(self._last_successful_recommendations.items()):

1640          Fallback is transparent to the caller.
1641  
1642          Args:
1643              messages: 对话消息列表

1646          Returns:

1672          total_tokens = sum(self.engine._count_tokens(m.get("content", "")) for m in messages) + max_tokens

1677  
1678          # 检查熔断器状态
1679          if not self.engine.circuit_breaker.can_execute():
1680              raise Exception("Circuit breaker is OPEN - LLM service unavailable")

1682          headers = {
1683              "Authorization": f"Bearer {self.engine.api_key}",
1684              "Content-Type": "application/json"
1685          }

1687          payload = {
1688              "model": self.engine.model,

1711                          err = RetryableError("Rate limit exceeded - please try again later")

1714                              try:
1715                                  err.retry_after = float(retry_after_header)

 247      """负载均衡器"""
 248  
 249      def __init__(self, strategy: str = "round_robin"):
 250          self.strategy = strategy
 251          self._backends: Dict[str, Dict] = {}
 252          self._current_index = 0
 253          self._lock = threading.Lock()
 254  
 255      def register_backend(
 256          self,
 257          backend_id: str,
 258          endpoint: str,
 259          weight: int = 1,
 260          max_connections: int = 100
 261      ):
 262          with self._lock:
 263              self._backends[backend_id] = {
 264                  "endpoint": endpoint,
 265                  "weight": weight,
 266                  "max_connections": max_connections,
 267                  "current_connections": 0,
 268                  "total_requests": 0,
 269                  "failed_requests": 0,
 270                  "avg_latency_ms": 0,
 271                  "healthy": True,
 272                  "last_check": datetime.now()
 273              }
 274  
 275      def get_backend(self) -> Optional[str]:
 276          with self._lock:
 277              healthy_backends = {
 278                  k: v for k, v in self._backends.items()
 279                  if v["healthy"] and v["current_connections"] < v["max_connections"]
 280              }
 281  
 282              if not healthy_backends:
 283                  return None
 284  
 285              if self.strategy == "round_robin":
 286                  backend_ids = list(healthy_backends.keys())
 287                  selected = backend_ids[self._current_index % len(backend_ids)]
 288                  self._current_index += 1
 289                  return selected
 290  
 291              elif self.strategy == "weighted":
 292                  total_weight = sum(v["weight"] for v in healthy_backends.values())
 293                  r = np.random.random() * total_weight
 294                  cumulative = 0
 295                  for backend_id, info in healthy_backends.items():
 296                      cumulative += info["weight"]
 297                      if r <= cumulative:
 298                          return backend_id
 299                  return list(healthy_backends.keys())[0]
 300  
 301              elif self.strategy == "least_connections":
 302                  return min(
 303                      healthy_backends.keys(),
 304                      key=lambda k: healthy_backends[k]["current_connections"]
 305                  )
 306  
 307              else:
 308                  return list(healthy_backends.keys())[0]
 309  
 310      def record_request_start(self, backend_id: str):
 311          with self._lock:
 312              if backend_id in self._backends:
 313                  self._backends[backend_id]["current_connections"] += 1
 314                  self._backends[backend_id]["total_requests"] += 1
 315  
 316      def record_request_end(
 317          self,
 318          backend_id: str,
 319          success: bool,
 320          latency_ms: float
 321      ):
 322          with self._lock:
 323              if backend_id in self._backends:
 324                  self._backends[backend_id]["current_connections"] -= 1
 325                  if not success:
 326                      self._backends[backend_id]["failed_requests"] += 1
 327  
 328                  current_avg = self._backends[backend_id]["avg_latency_ms"]
 329                  total = self._backends[backend_id]["total_requests"]
 330                  self._backends[backend_id]["avg_latency_ms"] = (
 331                      current_avg * (total - 1) + latency_ms
 332                  ) / total
 333  
 334      def mark_unhealthy(self, backend_id: str):
 335          with self._lock:
 336              if backend_id in self._backends:
 337                  self._backends[backend_id]["healthy"] = False
 338  
 339      def get_backend_stats(self) -> Dict[str, Dict]:
 340          with self._lock:
 341              return {
 342                  k: {
 343                      "endpoint": v["endpoint"],
 344                      "healthy": v["healthy"],
 345                      "current_connections": v["current_connections"],
 346                      "total_requests": v["total_requests"],
 347                      "failed_requests": v["failed_requests"],
 348                      "avg_latency_ms": v["avg_latency_ms"]
 349                  }
 350                  for k, v in self._backends.items()

 373          with self._lock:
 374              self._versions[algorithm_id][version] = {
 375                  "handler": handler,
 376                  "description": description,
 377                  "input_schema": input_schema or {},
 378                  "output_schema": output_schema or {},
 379                  "registered_at": datetime.now(),
 380                  "request_count": 0,
 381                  "avg_latency_ms": 0
 382              }
 383  
 384              if is_default or algorithm_id not in self._default_versions:
 385                  self._default_versions[algorithm_id] = version
 386  
 387      def get_handler(
 388          self,
 389          algorithm_id: str,
 390          version: Optional[str] = None
 391      ) -> Optional[Callable]:
 392          with self._lock:
 393              version = version or self._default_versions.get(algorithm_id)
 394              if not version:
 395                  return None
 396  
 397              version_info = self._versions.get(algorithm_id, {}).get(version)
 398              return version_info["handler"] if version_info else None
 399  
 400      def get_version_info(
 401          self,
 402          algorithm_id: str,
 403          version: Optional[str] = None
 404      ) -> Optional[Dict]:
 405          with self._lock:
 406              version = version or self._default_versions.get(algorithm_id)
 407              return self._versions.get(algorithm_id, {}).get(version)
 408  
 409      def list_algorithms(self) -> List[Dict]:
 410          with self._lock:
 411              return [
 412                  {
 413                      "algorithm_id": alg_id,
 414                      "default_version": self._default_versions.get(alg_id),
 415                      "available_versions": list(versions.keys())
 416                  }
 417                  for alg_id, versions in self._versions.items()
 418              ]
 419  
 420      def record_usage(
 421          self,
 422          algorithm_id: str,
 423          version: str,
 424          latency_ms: float
 425      ):
 426          with self._lock:
 427              if algorithm_id in self._versions and version in self._versions[algorithm_id]:
 428                  info = self._versions[algorithm_id][version]
 429                  info["request_count"] += 1
 430                  current_avg = info["avg_latency_ms"]
 431                  info["avg_latency_ms"] = (
 432                      current_avg * (info["request_count"] - 1) + latency_ms
 433                  ) / info["request_count"]

 436  class APIGateway:
 437      """API网关主类
 438  
 439      P1-11: Enhanced with route validation and default fallback algorithm.
 440      """
 441  
 442      # P1-11: Default fallback algorithms for each type
 443      DEFAULT_FALLBACK_ALGORITHMS = {
 444          AlgorithmType.BODY_METRICS: "body_metrics_v1",
 445          AlgorithmType.BODY_DETECTION: "body_detection_v1",
 446          AlgorithmType.STYLE_RECOGNITION: "style_recognition_v1",
 447          AlgorithmType.CLOTHING_SEGMENTATION: "segmentation_v1",
 448          AlgorithmType.RECOMMENDATION: "recommendation_v1",
 449          AlgorithmType.TREND_PREDICTION: "trend_prediction_v1",
 450          AlgorithmType.AESTHETIC_SCORING: "aesthetic_scoring_v1",
 451          AlgorithmType.VIRTUAL_TRYON: "virtual_tryon_v1",
 452          AlgorithmType.VISUAL_SEARCH: "visual_search_v1",
 453      }
 454  
 455      def __init__(self, config: APIConfig = None):
 456          self.config = config or APIConfig()
 457          self.rate_limiter = RateLimiter(
 458              max_requests=self.config.rate_limit_requests,
 459              window_seconds=self.config.rate_limit_window_seconds
 460          )
 461          self.authenticator = RequestAuthenticator()
 462          self.load_balancer = LoadBalancer()
 463          self.algorithm_registry = AlgorithmRegistry()
 464  
 465          self._request_queue: asyncio.Queue = None
 466          self._active_requests: Dict[str, APIRequest] = {}
 467          self._response_cache: Dict[str, Tuple[Any, datetime]] = {}
 468          self._stats: Dict[str, Any] = defaultdict(int)
 469          self._lock = threading.Lock()
 470  
 471      def register_algorithm(
 472          self,
 473          algorithm_id: str,
 474          handler: Callable,
 475          version: str = "v1",
 476          is_default: bool = True
 477      ):
 478          self.algorithm_registry.register_algorithm(
 479              algorithm_id=algorithm_id,
 480              version=version,
 481              handler=handler,
 482              is_default=is_default
 483          )
 484  
 485      async def process_request(
 486          self,
 487          algorithm: str,
 488          payload: Dict[str, Any],
 489          REDACTEDstr] = None,
 490          version: Optional[str] = None,
 491          priority: RequestPriority = RequestPriority.NORMAL
 492      ) -> APIResponse:
 493          start_time = time.time()
 494          request_id = self._generate_request_id()
 495  
 496          # P1-11: Validate algorithm route
 497          validated_algorithm = self._validate_algorithm_route(algorithm)
 498          if validated_algorithm is None:
 499              return APIResponse(
 500                  request_id=request_id,
 501                  success=False,
 502                  result=None,
 503                  error=f"Invalid algorithm route: {algorithm}. Valid routes: {[e.value for e in AlgorithmType]}",
 504                  execution_time_ms=0,
 505                  cache_hit=False,
 506                  model_version=version or "unknown",
 507                  timestamp=datetime.now()
 508              )
 509  
 510          if self.config.enable_authentication and api_key:
 511              auth_success, user_id = self.authenticator.authenticate(api_key, algorithm)
 512              if not auth_success:
 513                  return APIResponse(
 514                      request_id=request_id,
 515                      success=False,
 516                      result=None,
 517                      error=f"Authentication failed: {user_id}",
 518                      execution_time_ms=0,
 519                      cache_hit=False,
 520                      model_version=version or "unknown",
 521                      timestamp=datetime.now()
 522                  )
 523          else:
 524              user_id = None
 525  
 526          if self.config.enable_rate_limiting:
 527              client_id = user_id or api_key or "anonymous"
 528              allowed, rate_info = self.rate_limiter.check_rate_limit(client_id)
 529              if not allowed:
 530                  return APIResponse(
 531                      request_id=request_id,
 532                      success=False,
 533                      result=None,
 534                      error=f"Rate limit exceeded. Reset at {rate_info.reset_time}",
 535                      execution_time_ms=0,
 536                      cache_hit=False,
 537                      model_version=version or "unknown",
 538                      timestamp=datetime.now()
 539                  )
 540  
 541          cache_key = self._generate_cache_key(algorithm, payload, version)
 542          if self.config.enable_caching and cache_key in self._response_cache:
 543              cached_result, cached_time = self._response_cache[cache_key]
 544              if datetime.now() - cached_time < timedelta(seconds=self.config.cache_ttl_seconds):
 545                  self._stats["cache_hits"] += 1
 546                  return APIResponse(
 547                      request_id=request_id,
 548                      success=True,
 549                      result=cached_result,
 550                      error=None,
 551                      execution_time_ms=(time.time() - start_time) * 1000,
 552                      cache_hit=True,
 553                      model_version=version or "cached",
 554                      timestamp=datetime.now()
 555                  )
 556  
 557          handler = self.algorithm_registry.get_handler(algorithm, version)
 558          # P1-11: If handler not found, try default fallback algorithm
 559          if handler is None:
 560              fallback_id = self.DEFAULT_FALLBACK_ALGORITHMS.get(validated_algorithm)
 561              if fallback_id and fallback_id != algorithm:
 562                  handler = self.algorithm_registry.get_handler(fallback_id)
 563                  if handler is not None:
 564                      logger.warning(
 565                          "Algorithm %s not found, using fallback: %s",
 566                          algorithm, fallback_id
 567                      )
 568                      algorithm = fallback_id
 569  
 570          if handler is None:
 571              return APIResponse(
 572                  request_id=request_id,
 573                  success=False,
 574                  result=None,
 575                  error=f"Algorithm not found: {algorithm} (no fallback available)",
 576                  execution_time_ms=0,
 577                  cache_hit=False,
 578                  model_version=version or "unknown",
 579                  timestamp=datetime.now()
 580              )
 581  
 582          try:
 583              if asyncio.iscoroutinefunction(handler):
 584                  result = await handler(payload)
 585              else:
 586                  result = handler(payload)
 587  
 588              execution_time_ms = (time.time() - start_time) * 1000
 589  
 590              self.algorithm_registry.record_usage(algorithm, version or "v1", execution_time_ms)
 591  
 592              if self.config.enable_caching:
 593                  self._response_cache[cache_key] = (result, datetime.now())
 594  
 595              self._stats["successful_requests"] += 1
 596  
 597              return APIResponse(
 598                  request_id=request_id,
 599                  success=True,
 600                  result=result,
 601                  error=None,
 602                  execution_time_ms=execution_time_ms,
 603                  cache_hit=False,
 604                  model_version=version or "v1",
 605                  timestamp=datetime.now()
 606              )
 607  
 608          except Exception as e:
 609              self._stats["failed_requests"] += 1
 610              return APIResponse(
 611                  request_id=request_id,
 612                  success=False,
 613                  result=None,
 614                  error=str(e),
 615                  execution_time_ms=(time.time() - start_time) * 1000,
 616                  cache_hit=False,
 617                  model_version=version or "unknown",
 618                  timestamp=datetime.now()
 619              )
 620  
 621      def _generate_request_id(self) -> str:
 622          return hashlib.md5(f"{time.time()}{np.random.random()}".encode()).hexdigest()[:16]
 623  
 624      def _validate_algorithm_route(self, algorithm: str) -> Optional[AlgorithmType]:
 625          """P1-11: Validate that the algorithm route is a known AlgorithmType.
 626  
 627          Returns the corresponding AlgorithmType enum if valid, None otherwise.
 628          """
 629          try:
 630              return AlgorithmType(algorithm)
 631          except ValueError:
 632              # Also check by name (case-insensitive)
 633              for algo_type in AlgorithmType:
 634                  if algo_type.value.lower() == algorithm.lower():
 635                      return algo_type
 636              return None
 637  
 638      def _generate_cache_key(
 639          self,
 640          algorithm: str,
 641          payload: Dict,
 642          version: Optional[str]
 643      ) -> str:
 644          payload_str = json.dumps(payload, sort_keys=True)
 645          payload_hash = hashlib.md5(payload_str.encode()).hexdigest()
 646          return f"{algorithm}:{version or 'default'}:{payload_hash}"
 647  
 648      def get_stats(self) -> Dict[str, Any]:
 649          return {
 650              "total_requests": self._stats["successful_requests"] + self._stats["failed_requests"],
 651              "successful_requests": self._stats["successful_requests"],
 652              "failed_requests": self._stats["failed_requests"],
 653              "cache_hits": self._stats["cache_hits"],
 654              "cache_hit_rate": (
 655                  self._stats["cache_hits"] / 
 656                  (self._stats["successful_requests"] + self._stats["failed_requests"])
 657                  if (self._stats["successful_requests"] + self._stats["failed_requests"]) > 0
 658                  else 0
 659              ),
 660              "algorithms": self.algorithm_registry.list_algorithms(),
 661              "backends": self.load_balancer.get_backend_stats()
 662          }
 663  
 664      def clear_cache(self):
 665          self._response_cache.clear()
