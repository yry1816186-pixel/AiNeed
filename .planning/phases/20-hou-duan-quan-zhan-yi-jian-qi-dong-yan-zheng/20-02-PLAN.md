Create `apps/backend/prisma/seeds/demo-recommendations.seed.ts`:

```typescript
import { PrismaClient } from "@prisma/client";

export async function seedDemoRecommendations(prisma: PrismaClient, userMap: any, itemMap: any) {
  const demoRecommendations = [];
  const days = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"];
  const occasions = [
    "日常通勤",
    "商务会议",
    "约会社交",
    "休闲周末",
    "运动健身",
    "校园生活",
    "派对聚会",
  ];

  for (let i = 0; i < 7; i++) {
    const date = new Date();
    date.setDate(date.getDate() + i);

    const rec = await prisma.recommendationBatch.create({
      data: {
        userId: userMap.demoUser?.id || Object.values(userMap)[0]?.id,
        date: date,
        occasion: occasions[i],
        is_demo: true,
        provider: "sandbox",
        items: {
          create: [
            { clothingItemId: Object.values(itemMap)[i % Object.keys(itemMap).length]?.id },
            { clothingItemId: Object.values(itemMap)[(i + 1) % Object.keys(itemMap).length]?.id },
          ],
        },
      },
    });
    demoRecommendations.push(rec);
  }

  return { demoRecommendations };
}
```

Note: Adapt the above to match the actual Prisma schema's RecommendationBatch and related models. Read schema.prisma to find the exact model names and fields.
