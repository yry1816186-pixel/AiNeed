   ```prisma
     tracking
     analytics

     body_metrics

     data_export
     account_deletion
   }
   ```

2. Change `UserConsent.consentType` from `String` to `ConsentType` enum

3. Add `ConsentRecord` model:

   ```prisma
   model ConsentRecord {
     id          String   @id @default(cuid())
     userId      String
     consentType ConsentType
     action      String   // "granted" | "revoked" | "updated"
     version     String
     details     String?  // JSON details about what was consented
     ipAddress   String?
     userAgent   String?
     createdAt   DateTime @default(now())
     user        User     @relation(fields: [userId], references: [id])

     @@index([userId, consentType])
     @@map("consent_records")
   }
   ```

4. Add `ConsentVersion` model for GB/T 45574-2025 versioning:

   ```prisma
   model ConsentVersion {
     id          String   @id @default(cuid())
     consentType ConsentType
     version     String
     content     String   // Full consent text
     isActive    Boolean  @default(true)
     createdAt   DateTime @default(now())

     @@unique([consentType, version])
     @@map("consent_versions")
   }
   ```

   </action>
