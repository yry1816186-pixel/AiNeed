  /**
   * 粗排候选穿搭 — 增强版 selectOutfitForDay
   * 加入 reuse_boost、反馈权重、排除列表过滤
   *
   * @param outfits 用户所有穿搭方案
   * @param date 日期
   * @param forecast 天气预报
   * @param sceneHint 场景提示
   * @param alreadySelectedIds 已选方案ID集合 (去重用)
   * @param userId 用户ID
   * @param excludedItemIds 排除的单品ID列表
   * @param reuseBoost 衣橱复用加成系数 (默认 0.3, D-19)
   */
  roughRankOutfits(
    outfits: OutfitWithItems[],
    date: Date,
    forecast: DailyForecast | null,
    sceneHint: string | null,
    alreadySelectedIds: Set<string>,
    userId: string,
    excludedItemIds: string[] = [],
    reuseBoost: number = 0.3,
    feedbackWeights: Record<string, number> = {},
  ): ScoredOutfit[] {
    // Filter out repeats
    const available = outfits.filter(o => !alreadySelectedIds.has(o.id));
    if (available.length === 0) return [];

    // Score each outfit using existing selectOutfitForDay logic
    const results: ScoredOutfit[] = [];
    for (const outfit of available) {
      // Check hard exclusion: any item in excludedItemIds?
      if (excludedItemIds.length > 0 && outfit.items) {
        const hasExcluded = outfit.items.some(
          (oi) => oi.clothing && excludedItemIds.includes(oi.clothing.id)
        );
        if (hasExcluded) continue;
      }

      // Base scoring using EXISTING selectOutfitForDay logic
      const fakeSet = new Set(alreadySelectedIds);
      fakeSet.add(outfit.id);
      const base = this.selectOutfitForDay(
        [outfit],
        date,
        forecast,
        sceneHint,
        fakeSet
      );
      if (!base) continue;

      // Calculate wardrobe reuse rate (D-18)
      const reuseRate = this.calculateWardrobeReuseRate(outfit, userId);

      // Apply reuse boost (D-19): boosted = base * (1 + reuseBoost * reuseRate)
      let boostedScore = base.score * (1 + reuseBoost * reuseRate);

      // Apply feedback dimension weights if available (D-02, D-10)
      // Feedback weights came from WeeklyFeedbackService.getUserFeedbackWeights()
      if (Object.keys(feedbackWeights).length > 0) {
        const dimAdjust = Object.values(feedbackWeights).reduce((sum, w) => sum + w, 0)
                          / Object.keys(feedbackWeights).length;
        boostedScore *= (1 + dimAdjust * 0.1); // max ±10% adjustment
      }

      results.push({
        outfit: base.outfit,
        score: Math.round(boostedScore * 100) / 100,
        breakdown: base.breakdown,
        reuseRate,
      });
    }

    results.sort((a, b) => b.score - a.score);
    return results;
  }

  /**
   * 计算衣橱复用率 (D-18)
   * wardrobe_reuse_rate = 推荐搭配中使用用户已有衣橱衣物的单品数 / 总搭配单品数
   */
  calculateWardrobeReuseRate(outfit: OutfitWithItems, userId: string): number {
    if (!outfit.items || outfit.items.length === 0) return 0;
    const totalItems = outfit.items.length;
    const ownedItems = outfit.items.filter(
      (item) => item.clothing?.userId === userId
    ).length;
    return Math.round((ownedItems / totalItems) * 100) / 100;
  }

  async regenerateWeek(
    userId: string,
    preserveLocked: boolean = true,
    excludeItemIds?: string[],
  ): Promise<WeeklyPlanResponseDto> {
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    // Delete unlocked future plans (D-14: preserve isLocked=true)
    if (preserveLocked) {
      await this.prisma.outfitPlan.deleteMany({
        where: { userId, plannedDate: { gte: today }, isLocked: false },
      });
    } else {
      await this.prisma.outfitPlan.deleteMany({
        where: { userId, plannedDate: { gte: today } },
      });
    }

    // Get locked plans to skip during regeneration
    const lockedPlans = preserveLocked
      ? await this.prisma.outfitPlan.findMany({
          where: { userId, plannedDate: { gte: today }, isLocked: true },
        })
      : [];

    const lockedDates = new Set(lockedPlans.map(p => p.plannedDate.toISOString().slice(0, 10)));

    // Regenerate non-locked days using full generateWeeklyPlan but skip locked dates
    const result = await this.generateWeeklyPlan(userId);

    // Merge locked plans back in
    for (const locked of lockedPlans) {
      const dateStr = locked.plannedDate.toISOString().slice(0, 10);
      const existingIdx = result.plans.findIndex(p => p.plannedDate === dateStr);
      if (existingIdx >= 0) {
        // Replace generated plan with locked plan data
        const lockedOutfit = locked.outfitId
          ? await this.prisma.outfit.findUnique({ where: { id: locked.outfitId } })
          : null;
        result.plans[existingIdx] = {
          ...result.plans[existingIdx],
          id: locked.id,
          outfitId: locked.outfitId,
          outfit: lockedOutfit ? {
            id: lockedOutfit.id, name: lockedOutfit.name, coverImage: lockedOutfit.coverImage,
            occasions: lockedOutfit.occasions, seasons: lockedOutfit.seasons,
            style: lockedOutfit.style, rating: lockedOutfit.rating ? parseFloat(String(lockedOutfit.rating)) : null,
          } : null,
          sceneTag: locked.sceneTag,
          isSpecialEvent: locked.isSpecialEvent,
          eventName: locked.eventName,
          wardrobe_reuse_rate: locked.scoreBreakup ? (locked.scoreBreakup as { total: number }).total : undefined,
        };
      }
    }

    return result;
  }

   * @param forecasts 7天天气预报
   * @param outfit 当天选中的穿搭方案
   * @param allWeekOutfits 本周所有穿搭方案 (用于重复检测)
   */
  generateRiskNotes(
    dayIndex: number,
    forecasts: DailyForecast[],
    outfit: OutfitForRisk | null,
    allWeekOutfits: OutfitForRisk[]
  ): string[] {
    const notes: string[] = [];
    if (!outfit) return notes;

    const forecast = forecasts[dayIndex];

    // 1. Weather temperature swing >10°C between adjacent days
    if (dayIndex > 0 && forecasts[dayIndex - 1] && forecast) {
      const yesterdayAvg =
        ((forecasts[dayIndex - 1].tempHigh ?? 0) + (forecasts[dayIndex - 1].tempLow ?? 0)) / 2;
      const todayAvg = ((forecast.tempHigh ?? 0) + (forecast.tempLow ?? 0)) / 2;
      const diff = Math.abs(todayAvg - yesterdayAvg);
      if (diff > 10) {
        notes.push(`明天温差${diff.toFixed(0)}°C，注意调整穿搭`);
      }
    }

    // 2. Repeat outfit warning — same outfit appearing elsewhere in week
    if (allWeekOutfits.length > 0) {
      const repeatCount = allWeekOutfits.filter(
        (o, i) => i !== dayIndex && o.id === outfit.id
      ).length;
      if (repeatCount > 0) {
        notes.push("本周已有类似穿搭，建议更换款式");
      }
    }

    // 3. Weekend formal mismatch
    if (forecast) {
      const date = new Date(forecast.date ?? "");
      const dayOfWeek = date.getDay(); // 0=Sun, 6=Sat
      if (dayOfWeek === 0 || dayOfWeek === 6) {
        const occasions = outfit.occasions ?? [];
        const isFormal = occasions.some((o) => ["面试", "商务", "正式", "晚宴"].includes(o));
        if (isFormal) {
          notes.push("周末推荐正式穿搭，可考虑休闲选择");
        }
      }
    }

    // 4. Rain/storm warning
    if (forecast?.condition) {
      const cond = forecast.condition;
      if (cond.includes("雨") || cond.includes("雪") || cond.includes("storm")) {
        notes.push("预报有雨，建议选择防水面料和防滑鞋");
      }
    }

    return notes;
  }
}

import { Injectable, Logger } from "@nestjs/common";
import type { Prisma } from "@prisma/client";
import { PrismaService } from "../../../../common/prisma/prisma.service";
import { SubmitFeedbackDto, FeedbackQueryDto, FeedbackResponseDto } from "../dto/feedback.dto";

/**
 * 反馈类型 → 影响维度的映射 (D-10)
 */
const FEEDBACK_DIMENSION_MAP: Record<string, string> = {
  too_hot: "weather",
  too_cold: "weather",
  too_formal: "occasion",
  too_casual: "occasion",
  wrong_style: "style",
  too_expensive: "budget",
};

/**
 * 衰减半衰期: 30 天 (D-12)
 * weight(t) = weight_initial * 0.5^(days_since / 30)
 */
const DECAY_HALF_LIFE_DAYS = 30;

@Injectable()
export class WeeklyFeedbackService {
  private readonly logger = new Logger(WeeklyFeedbackService.name);

  constructor(private prisma: PrismaService) {}

  /**
   * 提交反馈
   */
  async submitFeedback(userId: string, dto: SubmitFeedbackDto): Promise<FeedbackResponseDto> {
    // dont_want_this 硬排除: 添加到 OutfitPlan excludedItemIds (D-11)
    if (dto.feedbackType === "dont_want_this") {
      await this.addToExclusionList(userId, dto.planId, dto.outfitId);
    }

    const feedback = await this.prisma.weeklyPlanFeedback.create({
      data: {
        userId,
        planId: dto.planId,
        outfitId: dto.outfitId,
        feedbackType: dto.feedbackType as Prisma.EnumFeedbackTypeFieldUpdateOperationsInput,
        weight: 1.0,
      },
    });

    return {
      id: feedback.id,
      planId: feedback.planId,
      outfitId: feedback.outfitId,
      feedbackType: feedback.feedbackType,
      weight: feedback.weight,
      createdAt: feedback.createdAt.toISOString(),
    };
  }

  /**
   * 获取用户的反馈权重 (经衰减计算)
   * 返回: { [dimension]: adjustment_factor } — 用于 CalendarPlanService 粗排
   */
  async getUserFeedbackWeights(userId: string): Promise<Record<string, number>> {
    const feedbacks = await this.prisma.weeklyPlanFeedback.findMany({
      where: { userId },
      orderBy: { createdAt: "desc" },
    });

    const now = Date.now();
    const dimensionWeights: Record<string, number[]> = {};

    for (const fb of feedbacks) {
      const dim = FEEDBACK_DIMENSION_MAP[fb.feedbackType];
      if (!dim) continue; // like/dislike/dont_want_this 不直接映射维度

      const daysSince = (now - fb.createdAt.getTime()) / (1000 * 60 * 60 * 24);
      const decay = Math.pow(0.5, daysSince / DECAY_HALF_LIFE_DAYS);
      const effectiveWeight = fb.weight * decay;

      if (!dimensionWeights[dim]) dimensionWeights[dim] = [];
      dimensionWeights[dim].push(effectiveWeight);
    }

    // Aggregate per dimension: average of effective weights
    const result: Record<string, number> = {};
    for (const [dim, weights] of Object.entries(dimensionWeights)) {
      const sum = weights.reduce((a, b) => a + b, 0);
      const avg = sum / weights.length;
      // Convert to adjustment: negative feedback reduces dimension score
      result[dim] = 1 - Math.min(avg, 0.5); // cap at 50% reduction
    }

    return result;
  }

  /**
   * 查询用户反馈历史
   */
  async getFeedbackHistory(
    userId: string,
    query?: FeedbackQueryDto
  ): Promise<FeedbackResponseDto[]> {
    const feedbacks = await this.prisma.weeklyPlanFeedback.findMany({
      where: {
        userId,
        ...(query?.feedbackType ? { feedbackType: query.feedbackType as any } : {}),
      },
      orderBy: { createdAt: "desc" },
      take: query?.limit ?? 20,
    });

    const now = Date.now();
    return feedbacks.map((fb) => ({
      id: fb.id,
      planId: fb.planId,
      outfitId: fb.outfitId,
      feedbackType: fb.feedbackType,
      weight: this.calculateDecayWeight(fb.createdAt, now),
      createdAt: fb.createdAt.toISOString(),
    }));
  }

  /**
   * 计算单个反馈的衰减后权重
   * decay = 0.5^(days_since / 30)
   */
  calculateDecayWeight(createdAt: Date, now: number = Date.now()): number {
    const daysSince = (now - createdAt.getTime()) / (1000 * 60 * 60 * 24);
    const decay = Math.pow(0.5, daysSince / DECAY_HALF_LIFE_DAYS);
    return Math.round(decay * 1000) / 1000;
  }

  /**
   * 将穿搭方案加入用户排除列表 (D-11: dont_want_this 硬排除)
   * 最大排除数 50，FIFO 淘汰
   */
  private async addToExclusionList(
    userId: string,
    planId: string,
    outfitId: string
  ): Promise<void> {
    const plan = await this.prisma.outfitPlan.findUnique({ where: { id: planId } });
    if (!plan) return;

    const existingExcludes: string[] = (plan.excludedItemIds as string[]) ?? [];
    const outfit = await this.prisma.outfit.findUnique({
      where: { id: outfitId },
      include: { items: { include: { clothing: true } } },
    });
    if (!outfit?.items) return;

    const newItemIds = outfit.items
      .map((oi) => oi.clothing?.id)
      .filter((id): id is string => !!id && !existingExcludes.includes(id));

    const updatedExcludes = [...existingExcludes, ...newItemIds].slice(-50); // Max 50

    await this.prisma.outfitPlan.update({
      where: { id: planId },
      data: { excludedItemIds: updatedExcludes },
    });

    this.logger.log(
      `Added ${newItemIds.length} items to exclusion list for plan ${planId} (total: ${updatedExcludes.length})`
    );
  }
}
