**Endpoint 3: Lock day plan (整体锁定 — 不同于 D-15 的 per-item lock)**

```typescript
  @Post("plan/:date/lock-plan")
  @ApiOperation({
    summary: "锁定某天的穿搭方案（整体锁定整天的计划）",
    description: "将指定日期的 OutfitPlan.isLocked 设置为 true。重新生成整周时，isLocked=true 的天会被保留不被覆盖。注意：这是锁定整天的方案，不同于 regenerate 时的 lockItemIds（per-item 锁定）。",
  })
  @ApiParam({ name: "date", description: "日期 YYYY-MM-DD", example: "2026-04-20" })
  @ApiResponse({ status: 200, description: "锁定成功" })
  @ApiResponse({ status: 401, description: "未授权" })
  @ApiResponse({ status: 404, description: "该日期无穿搭计划" })
  async lockPlan(
    @Request() req: AuthenticatedRequest,
    @Param("date") date: string,
  ) {
    const dateRegex = /^\d{4}-\d{2}-\d{2}$/;
    if (!dateRegex.test(date)) {
      throw new NotFoundException("日期格式必须为 YYYY-MM-DD");
    }
    const parsedDate = new Date(date);
    const plan = await this.calendarPlanService.getPlanByDate(req.user.id, parsedDate);
    if (!plan) throw new NotFoundException("该日期无穿搭计划");

    await this.calendarPlanService.lockPlan(req.user.id, plan.id);
    return { success: true, message: "已锁定该天穿搭方案" };
  }
```

**Endpoint 4: Exclude item (D-16)**

```typescript
  @Post("plan/:date/exclude-item")
  @ApiOperation({
    summary: "排除某件单品",
    description: "将指定单品加入该天穿搭方案的排除列表。粗排阶段即过滤掉包含该单品的方案。",
  })
  @ApiParam({ name: "date", description: "日期 YYYY-MM-DD", example: "2026-04-20" })
  @ApiBody({ schema: { properties: { itemId: { type: "string", description: "要排除的单品 ID" } } } })
  @ApiResponse({ status: 200, description: "排除成功" })
  @ApiResponse({ status: 401, description: "未授权" })
  async excludeItem(
    @Request() req: AuthenticatedRequest,
    @Param("date") date: string,
    @Body("itemId") itemId: string,
  ) {
    const dateRegex = /^\d{4}-\d{2}-\d{2}$/;
    if (!dateRegex.test(date)) {
      throw new NotFoundException("日期格式必须为 YYYY-MM-DD");
    }
    if (!itemId) throw new NotFoundException("itemId 不能为空");

    const parsedDate = new Date(date);
    const plan = await this.calendarPlanService.getPlanByDate(req.user.id, parsedDate);
    if (!plan) throw new NotFoundException("该日期无穿搭计划");

    await this.calendarPlanService.excludeItem(req.user.id, plan.id, itemId);
    return { success: true, message: `已排除单品 ${itemId}` };
  }
```
