```typescript
import { Injectable, Logger } from "@nestjs/common";
import { ConfigService } from "@nestjs/config";
import axios from "axios";

export interface ScoreOutfitCandidate {
  item_id: string;
  name: string;
  category: string;
  color_primary: string;
  style_tags: string[];
  price: number;
  seasons: string[];
  occasions: string[];
  material?: string;
  brand?: string;
}

export interface ScoreOutfitsRequest {
  candidates: ScoreOutfitCandidate[][];
  userProfile: Record<string, unknown>;
  context: {
    occasion: string;
    temperature?: number;
    season: string;
  };
  budget?: number;
}

export interface ScoreOutfitsResultItem {
  plan_id: string;
  overall_score: number;
  color_harmony_score: number;
  style_consistency_score: number;
  body_fit_score: number;
  weather_score: number;
  occasion: string | null;
  season: string | null;
  provider: string;
}

@Injectable()
export class FullOutfitEngineClient {
  private readonly logger = new Logger(FullOutfitEngineClient.name);
  private readonly mlServiceUrl: string;
  private readonly mlApiKey: string;

  constructor(private configService: ConfigService) {
    this.mlServiceUrl = this.configService.get<string>("ML_SERVICE_URL", "http://ai-service:8000");
    this.mlApiKey = this.configService.get<string>("ML_API_KEY", "xuno-dev-key");
  }

  /**
   * 精排候选穿搭方案 — 调用 FullOutfitEngine /score-outfits 端点
   *
   * @param candidates 粗排选出的候选方案列表 (每个方案是一组单品)
   * @param userProfile 用户全量画像 (bodyType, colorSeason, stylePreferences, priceRange 等)
   * @param context 当天场景上下文 (occasion, temperature, season)
   * @param budget 预算上限 (默认 5000)
   * @returns 按 overall_score 降序排列的精排结果
   * @throws 超时/网络错误时抛出，由调用方 (CalendarPlanService) 处理降级
   */
  async scoreOutfits(
    candidates: ScoreOutfitCandidate[][],
    userProfile: Record<string, unknown>,
    context: { occasion: string; temperature?: number; season: string },
    budget: number = 5000.0
  ): Promise<ScoreOutfitsResultItem[]> {
    try {
      const response = await axios.post<{
        success: boolean;
        ranked_outfits: ScoreOutfitsResultItem[];
        provider: string;
      }>(
        `${this.mlServiceUrl}/api/stylist/score-outfits`,
        {
          candidates,
          user_profile: userProfile,
          scene_context: {
            occasion: context.occasion,
            temperature: context.temperature,
            season: context.season,
          },
          budget,
        },
        {
          timeout: 5000,
          headers: {
            "Content-Type": "application/json",
            "X-ML-API-Key": this.mlApiKey,
          },
        }
      );

      this.logger.log(
        `FullOutfitEngine scored ${
          response.data.ranked_outfits?.length ?? 0
        } candidates (provider: ${response.data.provider})`
      );
      return response.data.ranked_outfits ?? [];
    } catch (error) {
      const errMsg = error instanceof Error ? error.message : String(error);
      this.logger.warn(`FullOutfitEngine unavailable: ${errMsg}`);
      throw error; // Let caller (CalendarPlanService) handle fallback to rough rank
    }
  }
}
```

**Key design decisions (per CONTEXT.md decisions):**

- Timeout: 5000ms (D-03: agent discretion, confirmed 5s in CONTEXT §Agent's Discretion)
- Error handling: Re-throw error — caller (CalendarPlanService) implements fallback to rough rank (D-03)
- URL: `${ML_SERVICE_URL}/api/stylist/score-outfits` — matching FastAPI endpoint from Plan 23-01
- Auth: Uses `X-ML-API-Key` header same as existing `ai-stylist.service.ts` pattern
- Provider: Logs `provider` field from response for audit trail

**Dependencies:**

- `axios` — already in project package.json (used by ai-stylist.service.ts)
- `@nestjs/config` ConfigService — already in AiStylistModule imports
- No new npm packages needed
