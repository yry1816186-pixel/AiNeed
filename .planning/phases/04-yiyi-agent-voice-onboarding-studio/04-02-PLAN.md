    **Create tts.service.ts:**

    The TTS service calls the Python backend's edge-tts endpoint (which will be added by Plan 01 or runs independently). Since edge-tts is a Python library, the actual audio generation happens in Python. NestJS acts as the gateway.

    ```typescript
    @Injectable()
    export class EdgeTTSService {
      private readonly logger = new Logger(EdgeTTSService.name);
      private readonly pythonBaseUrl: string;

      constructor(private readonly prisma: PrismaService) {
        this.pythonBaseUrl = process.env.PYTHON_AI_URL || "http://localhost:8000";
      }

      async synthesizeAndUpload(text: string, voice = "zh-CN-XiaoxiaoNeural"): Promise<string> {
        // Call Python FastAPI /tts/synthesize endpoint
        // Python returns audio file path or uploads to MinIO
        // NestJS returns the URL to the mobile client
        try {
          const response = await fetch(`${this.pythonBaseUrl}/tts/synthesize`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ text, voice, rate: "-10%" }),
          });
          if (!response.ok) {
            throw new Error(`TTS synthesis failed: ${response.status}`);
          }
          const data = await response.json() as { audio_url: string };
          return data.audio_url;
        } catch (error) {
          this.logger.warn(`TTS synthesis failed: ${error instanceof Error ? error.message : String(error)}`);
          throw error;
        }
      }
    }
    ```

    **Add TTS endpoint to ai-stylist.controller.ts:**
    ```typescript
    @Post("tts")
    async synthesizeSpeech(@Body() body: { text: string }) {
      const audioUrl = await this.ttsService.synthesizeAndUpload(body.text);
      return { audioUrl };
    }
    ```
