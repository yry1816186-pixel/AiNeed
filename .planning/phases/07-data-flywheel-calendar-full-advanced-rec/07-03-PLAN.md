   ```typescript
   export interface DailyForecast {
     date: string;           // "2026-04-20"
     tempHigh: number;
     tempLow: number;
     condition: string;      // "晴" | "多云" | "小雨"
     icon: string;           // QWeather icon code
     textDay: string;
     textNight: string;
   }

   async get7DayForecast(latitude: number, longitude: number): Promise<DailyForecast[]> {
     const url = `${this.qweatherBaseUrl}/weather/7d?location=${longitude},${latitude}&key=${this.qweatherApiKey}`;
     const response = await fetch(url);
     const data = await response.json();

     if (data.code !== '200') {
       this.logger.warn(`QWeather 7-day forecast failed: ${data.code}`);
       return [];
     }

     return (data.daily || []).map((day: any) => ({
       date: day.fxDate,
       tempHigh: parseInt(day.tempMax, 10),
       tempLow: parseInt(day.tempMin, 10),
       condition: day.textDay,
       icon: day.iconDay,
       textDay: day.textDay,
       textNight: day.textNight,
     }));
   }
   ```

   Use Redis caching with key `weather:7day:${lat}:${lon}` and TTL 3600 (1 hour).

3. **Create CalendarPlanService** at `apps/backend/src/domains/ai-core/ai-stylist/services/calendar-plan.service.ts`:

   @Injectable()
   export class CalendarPlanService {
     private readonly logger = new Logger(CalendarPlanService.name);

     constructor(
       private readonly prisma: PrismaService,
       private readonly weatherService: WeatherService,
       private readonly orchestrator: RecommendationOrchestrator,
       private readonly redisService: RedisService,
     ) {}

     /**
      * Weekly plan auto-generation cron (per D-09).
      * Every Monday at 6AM.
      */
     @Cron('0 6 * * 1')
     async handleWeeklyPlanGeneration(): Promise<void> {
       const activeUsers = await this.prisma.user.findMany({
         where: { isActive: true },
         select: { id: true, profile: true },
       });

       for (const user of activeUsers) {
         try {
           await this.generateWeeklyPlan(user.id);
         } catch (error) {
           this.logger.error(`Failed to generate plan for user ${user.id}: ${error}`);
         }
       }
     }

     /**
      * Generate 7-day outfit plan for user (per D-09).
      * Combines: weather forecast + calendar events + wardrobe items.
      */
     async generateWeeklyPlan(userId: string): Promise<OutfitPlan[]> {
       // 1. Get user's wardrobe outfits
       const wardrobeOutfits = await this.prisma.outfit.findMany({
         where: { userId },
         include: { items: true },
       });

       // 2. Get 7-day weather forecast
       const profile = await this.prisma.userProfile.findUnique({ where: { userId } });
       const forecast = await this.weatherService.get7DayForecast(
         profile?.latitude ?? 39.9,
         profile?.longitude ?? 116.4
       );

       // 3. Get user's special events from AiStylistSession
       const upcomingEvents = await this.getUpcomingEvents(userId);

       // 4. Generate plan for each day
       const plans: OutfitPlan[] = [];
       for (let i = 0; i < 7; i++) {
         const date = new Date();
         date.setDate(date.getDate() + i);
         const dateStr = date.toISOString().split('T')[0];

         const weather = forecast.find(f => f.date === dateStr);
         const event = upcomingEvents.find(e => e.date === dateStr);

         // Select best outfit based on weather + scene
         const selectedOutfit = await this.selectOutfitForDay(
           userId, wardrobeOutfits, weather, event
         );

         // Upsert plan
         const plan = await this.prisma.outfitPlan.upsert({
           where: { userId_plannedDate: { userId, plannedDate: date } },
           create: {
             userId,
             plannedDate: date,
             outfitId: selectedOutfit?.id ?? null,
             weatherContext: weather ? {
               tempHigh: weather.tempHigh,
               tempLow: weather.tempLow,
               condition: weather.condition,
               icon: weather.icon,
             } : null,
             sceneTag: event?.scene ?? this.inferSceneFromWeather(weather),
             isSpecialEvent: !!event,
             eventName: event?.name ?? null,
             source: 'ai_generated',
           },
           update: {
             outfitId: selectedOutfit?.id ?? undefined,
             weatherContext: weather ? {
               tempHigh: weather.tempHigh,
               tempLow: weather.tempLow,
               condition: weather.condition,
               icon: weather.icon,
             } : undefined,
           },
         });
         plans.push(plan);
       }
       return plans;
     }

     /**
      * Edit a specific day's outfit (per D-10).
      * Returns previous outfit info for preference signal (CAL-05).
      */
     async editDayPlan(userId: string, date: string, newOutfitId: string): Promise<{ plan: OutfitPlan; previousOutfitId: string | null }> {
       const plannedDate = new Date(date);
       const existing = await this.prisma.outfitPlan.findUnique({
         where: { userId_plannedDate: { userId, plannedDate } },
       });

       const previousOutfitId = existing?.outfitId ?? null;

       const plan = await this.prisma.outfitPlan.upsert({
         where: { userId_plannedDate: { userId, plannedDate } },
         create: {
           userId, plannedDate,
           outfitId: newOutfitId,
           source: 'manual',
         },
         update: {
           outfitId: newOutfitId,
           source: 'manual',
         },
       });

       // Emit preference signal (CAL-05): user replaced outfit → preference data
       await this.emitPreferenceSignal(userId, previousOutfitId, newOutfitId);

       return { plan, previousOutfitId };
     }

     /**
      * Check outfit repeat (per D-11).
      * Returns days since last wear if >70% item overlap.
      */
     async checkRepeatOutfit(userId: string, outfitId: string): Promise<number | null> {
       const outfit = await this.prisma.outfit.findUnique({
         where: { id: outfitId },
         include: { items: true },
       });
       if (!outfit) return null;

       const itemIds = new Set(outfit.items.map(i => i.clothingItemId));
       if (itemIds.size === 0) return null;

       // Find recent outfits with >70% overlap
       const recentWorn = await this.prisma.outfit.findMany({
         where: {
           userId,
           id: { not: outfitId },
           lastWorn: { not: null, gte: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000) },
         },
         include: { items: true },
         orderBy: { lastWorn: 'desc' },
       });

       for (const recent of recentWorn) {
         const recentIds = new Set(recent.items.map(i => i.clothingItemId));
         const overlap = [...itemIds].filter(id => recentIds.has(id)).length;
         const overlapRatio = overlap / itemIds.size;
         if (overlapRatio > 0.7) {
           const daysSince = Math.floor((Date.now() - recent.lastWorn!.getTime()) / (24 * 60 * 60 * 1000));
           return daysSince;
         }
       }
       return null;
     }
