2. **Create OutfitDiaryService** at `apps/backend/src/domains/platform/recommendations/services/outfit-diary.service.ts`:

   ```typescript
   @Injectable()
   export class OutfitDiaryService {
     constructor(private readonly prisma: PrismaService) {}

     /**
      * Auto-record diary entry from behavior event (per D-12).
      * Called by DiaryEventListener when save/try_on_complete events occur.
      */
     async autoRecordFromEvent(
       userId: string,
       eventType: string,
       outfitId: string,
       metadata?: Record<string, unknown>
     ): Promise<OutfitDiary> {
       const satisfactionScore = this.inferSatisfaction(eventType); // Per D-14
       return this.prisma.outfitDiary.create({
         data: {
           userId,
           outfitId,
           date: new Date(),
           scene: (metadata?.scene as string) ?? null,
           weather: (metadata?.weather as string) ?? null,
           temperature: (metadata?.temperature as number) ?? null,
           source: "auto",
           satisfactionScore,
           outfitSnapshot: metadata?.outfitSnapshot ?? null,
         },
       });
     }

     /**
      * Get diary entries for a date range.
      */
     async getDiaryEntries(userId: string, startDate: Date, endDate: Date): Promise<OutfitDiary[]> {
       return this.prisma.outfitDiary.findMany({
         where: {
           userId,
           date: { gte: startDate, lte: endDate },
         },
         orderBy: { date: "desc" },
         include: { outfit: true },
       });
     }

     /**
      * Infer satisfaction from event type (per D-14).
      * save=1.0, try_on_complete=0.6, skip=-0.3, reject=-0.5
      */
     private inferSatisfaction(eventType: string): number {
       const scores: Record<string, number> = {
         outfit_save: 1.0,
         favorite: 1.0,
         purchase: 1.0,
         try_on_complete: 0.6,
         recommendation_click: 0.3,
         skip: -0.3,
         unfavorite: -0.5,
       };
       return scores[eventType] ?? 0.0;
     }
   }
   ```

     constructor(

       private readonly diaryService: OutfitDiaryService,
     ) {}

     /**
      * Weekly report generation cron (per D-13).
      * Every Sunday at 8PM.
      */
     @Cron('0 20 * * 0')
     async handleWeeklyReportGeneration(): Promise<void> {
       // Find all users with diary entries in the past week

     }

     /**
      * Generate 7-element weekly report (per D-13).
      */
     async generateWeeklyReport(userId: string): Promise<WeeklyReport> {
       const weekEnd = new Date();
       const weekStart = new Date(weekEnd);
       weekStart.setDate(weekStart.getDate() - 7);

       const entries = await this.diaryService.getDiaryEntries(userId, weekStart, weekEnd);

       // 1. Satisfaction: average of all satisfactionScore values
       const satisfaction = entries.reduce((sum, e) => sum + (e.satisfactionScore ?? 0), 0) / Math.max(entries.length, 1);

       // 2. Style distribution: count entries by scene
       const styleDistribution = this.computeStyleDistribution(entries);

       // 3. Trend summary: compare with previous week
       const trendSummary = await this.computeTrendSummary(userId, weekStart, entries);

       // 4. Evolution curve: daily style dimension scores
       const evolutionCurve = this.computeEvolutionCurve(entries);

       // 5. Scene coverage: count by scene type
       const sceneCoverage = this.computeSceneCoverage(entries);

       // 6. Color analysis: aggregate outfit colors
       const colorAnalysis = this.computeColorAnalysis(entries);

       // 7. Item reuse rate: how many items worn >1 time
       const itemReuseRate = this.computeItemReuseRate(entries);

       return this.prisma.weeklyReport.upsert({
         where: { userId_weekStart: { userId, weekStart } },
         create: {
           userId, weekStart, weekEnd,
           satisfaction, styleDistribution, trendSummary,
           evolutionCurve, sceneCoverage, colorAnalysis, itemReuseRate,
         },
         update: {
           satisfaction, styleDistribution, trendSummary,
           evolutionCurve, sceneCoverage, colorAnalysis, itemReuseRate,
         },
       });
     }

4. **Create DiaryEventListener** at `apps/backend/src/domains/platform/recommendations/services/diary-event-listener.service.ts`:
   This service listens for behavior events (via BehaviorTrackerService flush) and auto-creates diary entries:

   ```typescript
   @Injectable()
   export class DiaryEventListener implements OnModuleInit {
     constructor(
       private readonly diaryService: OutfitDiaryService,
       private readonly prisma: PrismaService
     ) {}

     async onModuleInit() {
       // Subscribe to behavior events after flush
     }

     /**
      * Process recent behavior events and auto-create diary entries (per D-12).
      */
     async processRecentEvents(): Promise<number> {
       const triggerEvents = ["outfit_save", "try_on_complete", "favorite"];
       const recentEvents = await this.prisma.userBehaviorEvent.findMany({
         where: {
           eventType: { in: triggerEvents },
           userId: { not: null },
           targetId: { not: null },
         },
         orderBy: { createdAt: "desc" },
         take: 100,
       });

       let created = 0;
       for (const event of recentEvents) {
         if (!event.userId || !event.targetId) continue;
         // Check if diary entry already exists for this event
         const existing = await this.prisma.outfitDiary.findFirst({
           where: { userId: event.userId, outfitId: event.targetId, date: event.createdAt },
         });
         if (!existing) {
           await this.diaryService.autoRecordFromEvent(
             event.userId,
             event.eventType,
             event.targetId,
             event.metadata as Record<string, unknown>
           );
           created++;
         }
       }
       return created;
     }
   }
   ```
