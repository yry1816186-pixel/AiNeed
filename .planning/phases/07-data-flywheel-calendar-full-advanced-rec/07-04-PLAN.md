2. **Create CoordinationModel** at `ml/models/coordination_model.py`:

   ```python
   """
   Coordination Model: Dual-tower + Cross-attention for outfit compatibility scoring.

   Per D-17:
   - 10M params
   - Dual-tower: item A encoder + item B encoder (shared weights)
   - Cross-attention layer for fine-grained interaction
   - Output: compatibility score [0, 1]

   Architecture:
   - Item encoder: Embedding(50 categories, 128) -> FC(128->256) -> FC(256->128)
   - Cross-attention: Query=ItemA, Key=Value=ItemB, 4 heads, dim=128
   - Output head: FC(128*3 -> 64) -> FC(64 -> 1) -> Sigmoid
   """
   import torch
   import torch.nn as nn
   import math

   class ItemEncoder(nn.Module):
       """Encode a single fashion item into a fixed-size vector."""
       def __init__(self, num_categories: int = 50, embed_dim: int = 128, hidden_dim: int = 256):
           super().__init__()
           self.category_embed = nn.Embedding(num_categories, embed_dim)
           self.fc1 = nn.Linear(embed_dim + 16, hidden_dim)  # +16 for auxiliary features
           self.fc2 = nn.Linear(hidden_dim, embed_dim)
           self.layer_norm = nn.LayerNorm(embed_dim)
           self.relu = nn.ReLU()
           self.dropout = nn.Dropout(0.1)

       def forward(self, category_id: torch.Tensor, aux_features: torch.Tensor) -> torch.Tensor:
           cat_embed = self.category_embed(category_id)
           x = torch.cat([cat_embed, aux_features], dim=-1)
           x = self.relu(self.fc1(x))
           x = self.dropout(x)
           x = self.fc2(x)
           return self.layer_norm(x)

   class CrossAttention(nn.Module):
       """Cross-attention between two item encodings."""
       def __init__(self, dim: int = 128, num_heads: int = 4):
           super().__init__()
           self.attention = nn.MultiheadAttention(dim, num_heads, dropout=0.1, batch_first=True)
           self.layer_norm = nn.LayerNorm(dim)

       def forward(self, query: torch.Tensor, key_value: torch.Tensor) -> torch.Tensor:
           # query: (batch, 1, dim), key_value: (batch, 1, dim)
           attn_out, _ = self.attention(query, key_value, key_value)
           return self.layer_norm(query + attn_out)

   class CoordinationModel(nn.Module):
       """
       Full coordination model (~10M params).

       Dual-tower encoders (shared) + cross-attention + output head.
       """
       def __init__(
           self,
           num_categories: int = 50,
           embed_dim: int = 128,
           hidden_dim: int = 256,
           num_heads: int = 4,
       ):
           super().__init__()
           self.item_encoder = ItemEncoder(num_categories, embed_dim, hidden_dim)
           self.cross_attention_ab = CrossAttention(embed_dim, num_heads)
           self.cross_attention_ba = CrossAttention(embed_dim, num_heads)
           self.output_head = nn.Sequential(
               nn.Linear(embed_dim * 3, 128),
               nn.ReLU(),
               nn.Dropout(0.1),
               nn.Linear(128, 64),
               nn.ReLU(),
               nn.Linear(64, 1),
               nn.Sigmoid(),
           )

       def forward(
           self,
           cat_a: torch.Tensor,
           aux_a: torch.Tensor,
           cat_b: torch.Tensor,
           aux_b: torch.Tensor,
       ) -> torch.Tensor:
           # Encode both items
           embed_a = self.item_encoder(cat_a, aux_a).unsqueeze(1)  # (batch, 1, dim)
           embed_b = self.item_encoder(cat_b, aux_b).unsqueeze(1)

           # Cross-attention in both directions
           cross_ab = self.cross_attention_ab(embed_a, embed_b).squeeze(1)
           cross_ba = self.cross_attention_ba(embed_b, embed_a).squeeze(1)
           embed_a = embed_a.squeeze(1)

           # Concatenate and predict
           combined = torch.cat([embed_a, cross_ab, cross_ba], dim=-1)
           return self.output_head(combined).squeeze(-1)  # (batch,)

       def count_parameters(self) -> int:
           return sum(p.numel() for p in self.parameters() if p.requires_grad)
   ```

   @Injectable()
   export class CoordinationModelService {
     private readonly logger = new Logger(CoordinationModelService.name);
     private readonly baseUrl: string;
     private readonly enabled: boolean;

     constructor(private configService: ConfigService) {
       this.baseUrl = this.configService.get<string>(
         "COORDINATION_MODEL_URL",
         "http://localhost:8101"
       );
       this.enabled =
         this.configService.get<string>("COORDINATION_MODEL_ENABLED", "false") === "true";
     }

     isEnabled(): boolean {
       return this.enabled;
     }

     /**
      * Predict compatibility score between two items (per D-17).
      */
     async predictCompatibility(
       itemACategory: string,
       itemBCategory: string,
       auxA?: number[],
       auxB?: number[]
     ): Promise<number | null> {
       if (!this.enabled) return null;
       try {
         const response = await fetch(`${this.baseUrl}/coordination/predict`, {
           method: "POST",
           headers: { "Content-Type": "application/json" },
           body: JSON.stringify({

             item_b_category: itemBCategory,
             item_b_aux: auxB ?? new Array(16).fill(0),
           }),

         return data.compatibility_score;
       } catch (error) {
         this.logger.warn("Coordination model predict failed", error);
         return null;
       }
     }
