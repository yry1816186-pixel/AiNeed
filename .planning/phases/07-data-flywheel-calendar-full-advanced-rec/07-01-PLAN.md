   ```typescript
   @Injectable()
   export class BehaviorEtlService {
     private readonly logger = new Logger(BehaviorEtlService.name);
     private readonly MAX_SEQ_LENGTH = 50; // Per D-06: SASRec default

     constructor(private readonly prisma: PrismaService) {}

     /**
      * Extract user behavior sequences for SASRec training (per D-02: full extraction).
      * Groups by userId, sorts by createdAt, truncates to max_seq_length.
      */
     async extractTrainingSequences(): Promise<UserSequence[]> {
       const events = await this.prisma.userBehaviorEvent.findMany({
         where: { userId: { not: null } },
         orderBy: [{ userId: "asc" }, { createdAt: "asc" }],
         select: {
           userId: true,
           targetId: true,
           eventType: true,
           createdAt: true,
           metadata: true,
         },
       });

       const grouped = new Map<string, SequenceEvent[]>();
       for (const event of events) {
         if (!event.userId || !event.targetId) continue;
         const existing = grouped.get(event.userId) || [];
         existing.push({
           itemId: event.targetId,
           eventType: event.eventType,
           timestamp: event.createdAt.getTime(),
           implicitScore: this.computeImplicitScore(event.eventType),
         });
         grouped.set(event.userId, existing);
       }

       return Array.from(grouped.entries()).map(([userId, events]) => ({
         userId,
         events: events.slice(-this.MAX_SEQ_LENGTH), // Truncate to most recent per D-06
       }));
     }

     /**
      * Compute implicit feedback score from event type (per D-14).
      */
     private computeImplicitScore(eventType: string): number {
       const scores: Record<string, number> = {
         purchase: 1.0,
         favorite: 1.0,
         outfit_save: 1.0,
         add_to_cart: 0.8,
         try_on_complete: 0.6,
         recommendation_click: 0.4,
         click: 0.3,
         item_view: 0.1,
         skip: -0.3,
         unfavorite: -0.5,
         remove_from_cart: -0.5,
       };
       return scores[eventType] ?? 0.0;
     }
   }
   ```

   ```typescript
   @Injectable()
   export class SasrecRetrainService implements OnModuleInit {
     private readonly logger = new Logger(SasrecRetrainService.name);

     constructor(
       private readonly behaviorEtl: BehaviorEtlService,
       private readonly sasrecClient: SASRecClientService,
       private readonly evaluator: RetrainingEvaluatorService,
       private readonly prisma: PrismaService,
     ) {}

     async onModuleInit() {
       this.logger.log('SasrecRetrainService initialized — monthly cron active');
     }

     /**
      * Monthly SASRec retraining (per D-01).
      * Cron: 1st of month at 2AM.
      */
     @Cron('0 2 1 * *')
     async handleMonthlyRetrain(): Promise<void> {
       this.logger.log('Starting monthly SASRec retraining...');
       try {
         const sequences = await this.behaviorEtl.extractTrainingSequences();
         if (sequences.length === 0) {
           this.logger.warn('No training sequences found, skipping retrain');
           return;
         }

         // Save current model version as backup before retraining
         await this.evaluator.saveCurrentVersion('sasrec');

         // Trigger SASRec training pipeline
         const response = await fetch('http://localhost:8100/pipeline/train', {
           method: 'POST',
           headers: { 'Content-Type': 'application/json' },
           body: JSON.stringify({
             sequences: sequences.map(s => ({
               user_id: s.userId,
               items: s.events.map(e => e.itemId),
             })),
           }),
         });

         if (!response.ok) {
           throw new Error(`SASRec training failed: ${response.status}`);
         }

         const result = await response.json();
         this.logger.log(`SASRec retraining complete: ${JSON.stringify(result.metrics)}`);

         // Evaluate new model (per D-04)
         await this.evaluator.evaluateAndMaybeRollback('sasrec', result.metrics);
       } catch (error) {
         this.logger.error('SASRec monthly retrain failed', error);
       }
     }

}
````

   ```typescript
   @Injectable()
   export class RetrainingEvaluatorService {
     private readonly logger = new Logger(RetrainingEvaluatorService.name);
     private readonly ROLLBACK_THRESHOLD = 0.05;  // 5% metric degradation triggers rollback
     private readonly RECALL_K = 10;

     constructor(private readonly prisma: PrismaService) {}

     /**
      * Save current model version as rollback point (per D-04).
      */
     async saveCurrentVersion(modelType: 'sasrec' | 'fashionsiglip'): Promise<void> {
       await this.prisma.modelVersion.create({
         data: {
           modelType,
           version: `v_${Date.now()}`,
           status: 'backup',
           metrics: {},
         },
       });
     }

     /**
      * Evaluate retrained model and rollback if degraded (per D-04, D-07).
      */
     async evaluateAndMaybeRollback(
       modelType: 'sasrec' | 'fashionsiglip',
       newMetrics: Record<string, number>,
     ): Promise<boolean> {
       const previousVersion = await this.prisma.modelVersion.findFirst({
         where: { modelType, status: 'active' },
         orderBy: { createdAt: 'desc' },
       });

       if (!previousVersion) {
         // No previous version, just activate new one
         await this.activateNewVersion(modelType, newMetrics);
         return false;
       }

       const prevMetrics = (previousVersion.metrics as Record<string, number>) ?? {};
       const prevRecall = prevMetrics[`recall@${this.RECALL_K}`] ?? 0;
       const newRecall = newMetrics[`recall@${this.RECALL_K}`] ?? 0;

       if (prevRecall > 0 && (prevRecall - newRecall) > this.ROLLBACK_THRESHOLD) {
         this.logger.warn(
           `${modelType} degraded: Recall@${this.RECALL_K} ${prevRecall} -> ${newRecall}. Rolling back.`
         );
         await this.rollback(modelType, previousVersion.version);
         return true;  // rolled back
       }

       await this.activateNewVersion(modelType, newMetrics);
       return false;  // no rollback
     }

     private async activateNewVersion(modelType: string, metrics: Record<string, number>): Promise<void> { ... }
     private async rollback(modelType: string, version: string): Promise<void> { ... }
   }
   ```
