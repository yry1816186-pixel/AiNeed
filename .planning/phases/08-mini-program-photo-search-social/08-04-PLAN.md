       ```typescript
       import Taro from '@tarojs/taro';
       import { upload, get } from './request';

       export async function searchByImage(tempFilePath: string, limit: number = 5) {
         const result = await upload<{ items: any[] }>(`/search/image?limit=${limit}`, tempFilePath, 'file');
         return result.items || result; // Handle both array and wrapped response
       }

       export async function searchByImageUrl(imageUrl: string, limit: number = 5) {
         return await post<any[]>(`/search/image/url?limit=${limit}`, { imageUrl });
       }

       export async function getSimilarItems(itemId: string, limit: number = 5) {
         return await get<any[]>(`/search/similar/${itemId}?limit=${limit}`);
       }
       ```
