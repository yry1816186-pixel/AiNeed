    7. **Create `src/services/auth.ts`**:
       ```typescript
       import Taro from '@tarojs/taro';
       import { post } from './request';

       export async function wechatMiniLogin(): Promise<{ accessToken: string; user: any }> {
         const { code } = await Taro.login();
         const res = await post<{ accessToken: string; refreshToken: string; user: any }>('/auth/wechat-mini', { code });
         Taro.setStorageSync('access_token', res.accessToken);
         Taro.setStorageSync('refresh_token', res.refreshToken);
         return { accessToken: res.accessToken, user: res.user };
       }

       export async function ensureLogin(): Promise<string | null> {
         const token = Taro.getStorageSync('access_token');
         if (token) return token;
         try {
           const result = await wechatMiniLogin();
           return result.accessToken;
         } catch {
           return null;
         }
       }
       ```
